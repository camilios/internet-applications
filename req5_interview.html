upstream req5_api {
    zone req5_api 64k;
    # Default: equal-weight round robin over the current Docker DNS records.
    server api:8000 resolve max_fails=1 fail_timeout=2s;
}
server {
    listen 80;
    client_max_body_size 24m;
    proxy_http_version 1.1;
    proxy_set_header Host $http_host;
    proxy_set_header Connection "";
    proxy_set_header X-Forwarded-For $remote_addr;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-Host $http_host;
    proxy_set_header X-Forwarded-Port 8015;
    proxy_set_header Forwarded "";
    proxy_connect_timeout 1s;
    proxy_next_upstream error timeout http_502 http_503 http_504;
    proxy_next_upstream_tries 3;
    add_header X-Req5-Upstream $upstream_addr always;

    location / {
        proxy_pass http://req5_api;
        error_page 502 = @retry_read;
    }

    location @retry_read {
        # Docker DNS can change the upstream membership between two attempts.
        # Only reads may restart selection after a gateway-generated 502.
        if ($request_method !~ ^(GET|HEAD)$) { return 502; }
        set $req5_read_target api:8000;
        proxy_pass http://$req5_read_target;
    }
}

ماذا يحدث عندما نوقف النسخة الأولى أثناء الـ٥٠٠ طلب؟

افتح الدالة live_load(). فيها ١٢ عميلاً يرسلون GET /up إلى البوابة. عند اكتمال ١٠٠ نتيجة، تعطي trigger.set() إشارة لمهمة مستقلة تنفّذ docker stop -t 2 على النسخة الأولى. تواصل بقية المهام إرسال الطلبات. وبعد أن يؤكد docker inspect أن الحاوية توقفت، يرفع الاختبار الإشارة down. آخر ٦٠ طلباً تنتظر هذه الإشارة عمداً، حتى نعرف يقيناً أن الزبائن استخدموا الخدمة والنسخة متوقفة.

في Nginx، تسمح proxy_next_upstream بتجربة عنوان آخر إذا فشل الاتصال، انتهت مهلته، أو رجعت إحدى الحالات المضبوطة مثل 502. تحدد proxy_next_upstream_tries 3 سقف محاولات اختيار نسخة في كل مرحلة. وعند تسجيل إخفاق وفق max_fails=1 fail_timeout=2s يتجنب Nginx العنوان مؤقتاً. إذا انتهت محاولة الاتصال بخطأ 502 من البوابة أثناء تغيّر عناوين النسخ، يعيد المسار @retry_read اختيار عنوان من خدمة Docker لطلبات القراءة GET وHEAD فقط. تبقى طريقة الطلب ومساره والترويسات كما هي. العميل نفسه يرسل كل طلب مرة واحدة، لذلك نحسب أي خطأ وصل إليه ضمن النتيجة.

ops/req5/interview.py · الأسطر 65–124
def live_load(first, name, peers):
    trigger, stopping, down = Event(), Event(), Event()
    lock = Lock()
    counts = Counter()
    failures = Counter()
    rows = []
    started = time.monotonic()
    marks = {}

    def stop_one():
        if not trigger.wait(60):
            raise RuntimeError('Load did not reach the stop threshold')
        marks['stop_requested_s'] = time.monotonic() - started
        stopping.set()
        v.command('docker','stop','-t','2',first)
        v.check(v.command('docker','inspect','-f','{{.State.Running}}',first)=='false','Replica still running')
        marks['stop_confirmed_s'] = time.monotonic() - started
        down.set()

    def one(i):
        # Keep the same pool active through the stop, and reserve a tail issued
        # only after Docker confirms it. No client retries or hidden exclusions.
        if i >= 440 and not down.wait(45):
            raise RuntimeError('Stop not confirmed during load')
        time.sleep(0.06)
        phase = 'down' if down.is_set() else 'transition' if stopping.is_set() else 'before'
        try:
            code, ip, _ = v.http('/up')
            who = peers.get(ip)
            ok = code == 200 and who is not None
            error = str(code) if code != 200 else 'unknown upstream'
        except (OSError, URLError) as exc:
            who, ok, error = None, False, type(exc).__name__
        with lock:
            if ok:
                counts[phase,who] += 1
            else:
                failures[error] += 1
            rows.append((phase,ok))
            if len(rows) == 100:
                trigger.set()

    try:
        with futures.ThreadPoolExecutor(max_workers=1) as control, futures.ThreadPoolExecutor(max_workers=12) as clients:
            stop_result = control.submit(stop_one)
            list(clients.map(one,range(500)))
            stop_result.result()
        results = {p:{w:c for (ph,w),c in counts.items() if ph==p}
                   for p in ('before','transition','down')}
        stats = {'attempted':len(rows),'http_200':sum(counts.values()),'errors':dict(failures),
                 'error_percent':100*sum(failures.values())/len(rows),'replicas':results,'stop_target':name,
                 'elapsed_s':round(time.monotonic()-started,3),**{k:round(val,3) for k,val in marks.items()}}
        print('500 requests / 12 clients:',json.dumps(stats),flush=True)
        v.check(len(rows)==500 and sum(failures.values())<=2,'More than 2 user-visible errors in 500 requests')
        v.check(set(results['before'].keys())==set(peers.values()),'Not all three served before the stop')
        v.check(set(results['down'].keys())==set(peers.values())-{name},'Unexpected distribution after confirmed stop')
        v.check(sum(results['transition'].values())>0,'No successful requests observed during the stop command')
        return stats
    finally:
        start_healthy(first)

    public function receive(): ?IncomingMessage
    {
        // Polling keeps heartbeat/socket handling bounded, including on an idle queue.
        $message = $this->channel()->basic_get(config('rabbitmq.queue'), false);
        if ($message === null) {
            return null;
        }
        $headers = $message->has('application_headers') ? $message->get('application_headers')->getNativeData() : [];
        $attempt = filter_var($headers['attempt'] ?? 1, FILTER_VALIDATE_INT);

        return new IncomingMessage(
            $message->getBody(),
            $attempt === false ? 1 : max(1, $attempt),
            fn () => $message->ack(),
        );
    }

    public function send(array $event): void
    {
        $mode = $this->mailConfiguration->assertDeliverable();

        $outbox = OutboxEvent::query()->useWritePdo()->find($event['event_id']);
        if ($outbox === null || $outbox->event_type !== 'OrderCreated' || $outbox->order_id !== $event['order_id']) {
            throw new PermanentMessageException('Message does not identify a recorded order event.');
        }

        // Unique row + row lock serializes duplicate deliveries, even across worker processes.
        DB::table('notification_deliveries')->insertOrIgnore([
            'event_id' => $event['event_id'], 'order_id' => $event['order_id'],
            'created_at' => now(), 'updated_at' => now(),
        ]);
        DB::transaction(function () use ($event, $mode): void {
            $delivery = NotificationDelivery::query()->whereKey($event['event_id'])->lockForUpdate()->firstOrFail();
            if ($delivery->delivered_at !== null) {
                // A provider change does not automatically replay earlier mock or unclassified deliveries.
                return;
            }

            $order = Order::query()->useWritePdo()->with(['user', 'items.product'])->find($event['order_id']);
            if ($order === null || $order->user === null || ! filter_var($order->user->email, FILTER_VALIDATE_EMAIL)) {
                throw new PermanentMessageException('Order recipient is unavailable.');
            }

            $sent = Mail::to($order->user->email)->send(new OrderConfirmation($order, $event['event_id']));
            if ($sent === null) {
                throw new RuntimeException('Mail delivery was cancelled before sending.');
            }
            // SMTP cannot participate in this transaction. A crash here can resend; stable Message-ID helps providers deduplicate.
            $delivery->update(['delivered_at' => now(), 'delivery_mode' => $mode]);
            Log::info($mode === 'mock' ? 'MOCK DELIVERY: order confirmation processed without an external email' : 'Order confirmation accepted by delivery transport', [
                'event_id' => $event['event_id'], 'delivery_mode' => $mode,
                'transport_message_id' => substr(preg_replace('/[^A-Za-z0-9@._:+\/-]/', '', $sent->getMessageId()), 0, 255),
            ]);
        }, 1);
    }

def inbox_counts():
    query = urlencode({'query':'to:'+RECIPIENT,'limit':100})
    with v.opener.open(INBOX+'/api/v1/search?'+query,timeout=8) as response:
        box = json.load(response)
    # Count distinct actual mailbox entries by order; never use Message-ID as a
    # unique count, since a duplicate may have the exact same Message-ID.
    ids, mail = set(), Counter()
    for msg in box['messages']:
        if not any(x['Address']==RECIPIENT for x in msg['To']):
            continue
        v.check(msg['ID'] not in ids,'Mailbox returned the same entry twice')
        ids.add(msg['ID'])
        m = re.fullmatch(r'Order #(\d+) confirmation',msg['Subject'])
        v.check(bool(m),'Unexpected email to the run-specific recipient')
        mail[int(m[1])] += 1
    return mail

صور المنتجاتالصور تُكتب في storage/app/public. كان المشروع يستخدم وحدة تخزين لهذا المسار أصلاً.تركب النسخ نفس الوحدة public_uploads. ملف التجربة الذي كتبته النسخة الأولى ظهر في الثلاث على مضيف Docker نفسه.
هوية المستخدم والكاشمفاتيح JWT والكاش كانت مشتركة أصلاً. أي اختلاف في المفاتيح بين النسخ يجعل رمز الزبون يفشل عند انتقاله.ترث النسخ مفاتيح التطبيق نفسها وتشارك Redis للكاش والجلسات. نجح طلب /api/me بنفس الرمز عبر النسخ، وبعد تسجيل الخروج أعادت الثلاث 401.

ملفات Blade المترجمة وكاش الإعدادات يمكن لكل حاوية إعادة توليدها. أمّا الطلبات وسجلات التسليم فتحفظها MySQL، ويحتفظ RabbitMQ بالرسائل. نطاق العرض هو توقف حاوية تطبيق على مضيف واحد؛ ما زال المضيف والبوابة وقاعدة البيانات نقاطاً تعتمد عليها النسخ الثلاث.

للمقارنة بين إعداد النسخة الواحدة وإعداد التوسّع وقراءة نتائج فحص التخزين افتح إعداد المشروع الأصلي، إعداد النسخ الثلاث، وتقرير فحص الحالة المشتركة.

جهاز الزميل مع Docker Desktop

من PowerShell في مشروع الزميل، تأكد من وجود ملف البيئة ومجلد الأسرار المحليين. نفّذ كل سطر كاملاً وانتظر نجاحه. إذا كان بناء الصور جارياً بالفعل، تابع من سطر التشغيل بعد اكتماله.

Set-Location 'C:\Users\NEW MSI\Desktop\see'
git pull --ff-only origin main
docker compose -f docker-compose.req5.yml -f docker-compose.req5-interview.yml build migrate inventory-service
docker compose -f docker-compose.req5.yml -f docker-compose.req5-interview.yml up -d --no-build --wait
docker compose -f docker-compose.req5.yml -f docker-compose.req5-interview.yml ps api email-worker
py -3 -X utf8 .\ops\req5\interview.py

في أول تحضير يبني Docker صورة Laravel التي تتضمن امتداد gRPC. إذا بلغ سجل البناء حد العرض، اترك الأمر حتى يعود مؤشر PowerShell وافحص $LASTEXITCODE. الصفر يعني نجاح البناء. عند تجهيز الخدمات مسبقاً يكفي تشغيل سطر الاختبار الأخير من مجلد المشروع.
