    public function createOrder(User $user, array $items, string $idempotencyKey): Order
    {
        $normalizedItems = collect($items)
            ->map(fn (array $item) => [
                'product_id' => (int) $item['product_id'],
                'quantity' => (int) $item['quantity'],
            ])
            ->sortBy(['product_id', 'quantity'])
            ->values()
            ->all();
        $requestHash = hash('sha256', json_encode($normalizedItems, JSON_THROW_ON_ERROR));

        DB::table('inventory_operations')->insertOrIgnore([
            'idempotency_key' => $idempotencyKey,
            'user_id' => $user->id,
            'request_hash' => $requestHash,
            'status' => 'pending',
            'created_at' => now(),
            'updated_at' => now(),
        ]);
        $operation = InventoryOperation::where('idempotency_key', $idempotencyKey)->firstOrFail();

        if ($operation->user_id !== $user->id || ! hash_equals($operation->request_hash, $requestHash)) {
            throw new \InvalidArgumentException('This Idempotency-Key was already used for a different order.');
        }
        if ($operation->status === 'completed') {
            return Order::with('items.product')->findOrFail($operation->order_id);
        }
        if ($operation->status === 'compensation_pending') {
            throw new InventoryOutcomeUnknownException('A prior compensation is still pending.');
        }
        if ($operation->status === 'compensated') {
            throw new \InvalidArgumentException('The previous attempt was compensated. Use a new Idempotency-Key.');
        }

        $productIds = array_column($items, 'product_id');
        $products = Product::whereIn('id', $productIds)
            ->where('is_active', true)
            ->get()
            ->keyBy('id');

        // Check all products exist and are active
        foreach ($items as $item) {
            if (! $products->has($item['product_id'])) {
                throw new \InvalidArgumentException(
                    "Product ID {$item['product_id']} not found or is inactive."
                );
            }
        }

        $orderItems = [];
        $total = '0.00';
        foreach ($items as $item) {
            $product = $products[$item['product_id']];
            $unitPrice = $product->price;
            $subtotal = bcmul($unitPrice, (string) $item['quantity'], 2);
            $total = bcadd($total, $subtotal, 2);

            $orderItems[] = [
                'product_id' => $product->id,
                'quantity' => $item['quantity'],
                'unit_price' => $unitPrice,
                'subtotal' => $subtotal,
            ];
        }

        $reservationId = null;
        try {
            $reservedOperation = DB::transaction(function () use (
                $operation,
                $items,
                $idempotencyKey
            ) {
                $lockedOperation = InventoryOperation::query()
                    ->whereKey($operation->id)
                    ->lockForUpdate()
                    ->firstOrFail();

                if ($lockedOperation->status === 'completed') {
                    return $lockedOperation;
                }
                if ($lockedOperation->status === 'compensation_pending') {
                    throw new InventoryOutcomeUnknownException('A prior compensation is still pending.');
                }
                if ($lockedOperation->status === 'compensated') {
                    throw new \InvalidArgumentException('The previous attempt was compensated. Use a new Idempotency-Key.');
                }

                if ($lockedOperation->status === 'pending') {
                    $reservation = $this->inventoryClient->reserveStock($items, $idempotencyKey);
                    if (! $reservation->success) {
                        throw new InsufficientStockException(
                            $reservation->reason ?: 'Insufficient stock for one or more products.'
                        );
                    }
                    if (! $reservation->reservationId) {
                        throw new InventoryOutcomeUnknownException('Reservation was not confirmed with an ID.');
                    }

                    $lockedOperation->update([
                        'status' => 'reserved',
                        'reservation_id' => $reservation->reservationId,
                    ]);
                }

                if ($lockedOperation->status !== 'reserved' || ! $lockedOperation->reservation_id) {
                    throw new InventoryOutcomeUnknownException('Reservation operation is not in a recoverable state.');
                }

                return $lockedOperation->fresh();
            });
            $reservationId = $reservedOperation->reservation_id;

            if ($reservedOperation->status === 'completed') {
                return Order::with('items.product')->findOrFail($reservedOperation->order_id);
            }

            $order = DB::transaction(function () use ($operation, $user, $total, $orderItems) {
                $lockedOperation = InventoryOperation::query()
                    ->whereKey($operation->id)
                    ->lockForUpdate()
                    ->firstOrFail();

                if ($lockedOperation->status === 'completed') {
                    return Order::with('items.product')->findOrFail($lockedOperation->order_id);
                }
                if ($lockedOperation->status !== 'reserved' || ! $lockedOperation->reservation_id) {
                    throw new InventoryOutcomeUnknownException('Confirmed reservation state is unavailable.');
                }

                $order = Order::create([
                    'user_id' => $user->id,
                    'total' => $total,
                    'status' => 'pending',
                    'reservation_id' => $lockedOperation->reservation_id,
                ]);

                foreach ($orderItems as $item) {
                    $order->items()->create($item);
                }

                $lockedOperation->update([
                    'status' => 'completed',
                    'order_id' => $order->id,
                    'last_error' => null,
                ]);
                $eventId = (string) Str::uuid();
                $occurredAt = now()->toISOString();
                OutboxEvent::create([
                    'event_id' => $eventId,
                    'order_id' => $order->id,
                    'event_type' => 'OrderCreated',
                    'payload' => ['event_id' => $eventId, 'type' => 'OrderCreated', 'order_id' => $order->id, 'occurred_at' => $occurredAt],
                    'available_at' => now(),
                ]);
                $this->notifyAfterCommit($user, $order, 'order_created');
                $this->broadcastAfterCommit($order, new OrderCreated($eventId, $order->id, $occurredAt));

                return $order;
            });
        } catch (InsufficientStockException|InventoryUnavailableException $exception) {
            throw $exception;
        } catch (\Throwable $e) {
            if ($reservationId === null) {
                try {
                    $durableOperation = InventoryOperation::find($operation->id);
                    if ($durableOperation?->status === 'completed') {
                        return Order::with('items.product')->findOrFail($durableOperation->order_id);
                    }
                    $reservationId = $durableOperation?->reservation_id;
                } catch (\Throwable $lookupException) {
                    Log::critical('Could not determine whether an order reservation was persisted', [
                        'operation_id' => $operation->id,
                        'error' => $lookupException->getMessage(),
                    ]);
                    throw new InventoryOutcomeUnknownException('Order persistence outcome is not known.');
                }
            }

            if ($reservationId === null) {
                throw $e;
            }

            try {
                $durableOperation = InventoryOperation::find($operation->id);
                if ($durableOperation?->status === 'completed') {
                    return Order::with('items.product')->findOrFail($durableOperation->order_id);
                }
            } catch (\Throwable $lookupException) {
                Log::critical('Could not verify order persistence before compensation', [
                    'operation_id' => $operation->id,
                    'reservation_id' => $reservationId,
                    'error' => $lookupException->getMessage(),
                ]);
                throw new InventoryOutcomeUnknownException('Order persistence outcome is not known.');
            }

            Log::error('Order creation failed, attempting stock release', [
                'reservation_id' => $reservationId,
                'error' => $e->getMessage(),
            ]);

            $released = false;
            $compensationError = null;
            try {
                $released = $this->inventoryClient->releaseStock($reservationId);
                if (! $released) {
                    $compensationError = 'Inventory service did not confirm the release.';
                }
            } catch (\Throwable $releaseException) {
                $compensationError = $releaseException->getMessage();
            }

            DB::transaction(function () use ($operation, $reservationId, $released, $compensationError, $e) {
                $lockedOperation = InventoryOperation::query()
                    ->whereKey($operation->id)
                    ->lockForUpdate()
                    ->firstOrFail();
                $lockedOperation->update([
                    'status' => $released ? 'compensated' : 'compensation_pending',
                    'reservation_id' => $reservationId,
                    'last_error' => $released ? $e->getMessage() : $compensationError,
                ]);

                if (! $released) {
                    FailedCompensation::firstOrCreate(
                        ['operation_id' => $lockedOperation->id],
                        [
                            'reservation_id' => $reservationId,
                            'status' => 'pending',
                            'error_message' => $compensationError,
                            'attempts' => 1,
                            'last_attempted_at' => now(),
                        ]
                    );
                }
            });

            if (! $released) {
                Log::critical('Stock compensation recorded for retry', [
                    'operation_id' => $operation->id,
                    'reservation_id' => $reservationId,
                    'error' => $compensationError,
                ]);
            }

            throw $e;
        }

        return $order->load('items.product');
    }

    private function broadcastAfterCommit(Order $order, OrderCreated|OrderStatusUpdated $event): void
    {
        DB::afterCommit(function () use ($order, $event): void {
            try {
                DashboardCache::forgetCache($order->user_id);
            } catch (\Throwable $exception) {
                Log::error('Order dashboard cache invalidation failed', ['order_id' => $order->id, 'error_type' => $exception::class]);
            }
            try {
                event($event);
            } catch (\Throwable $exception) {
                // A committed order must not be compensated due to a secondary broadcast failure.
                Log::error('Order broadcast enqueue failed after commit', ['order_id' => $order->id, 'error_type' => $exception::class]);
            }
        });
    }

ما تقوله

تخيّل لوحة المسؤول مفتوحة على إجمالي عشرة طلبات. اشترى زبون من متصفح آخر، ونريد أن يرى المسؤول أحد عشر من دون الضغط على تحديث. داخل دالة الإنشاء أذهب مباشرة إلى الحدث المُعدّ للبث بعد حفظ الطلب. في نفس المنطقة يوجد سجل الأحداث الذي يقرؤه ناشر RabbitMQ، لكن هذا ليس ما تشترك به لوحة المسؤول. هناك عملان انطلقا من واقعة شراء واحدة: تأكيد بريد يصل إلى الزبون، وإشارة بث تصل للإدارة؛ لا تنتظر الواحدة منهما انتهاء الأخرى.

في الدالة التالية يظهر سبب تأجيل الإشارة. لا تُرسل الدعوة إلى إعادة جلب القائمة قبل أن يصبح الطلب قابلاً للقراءة بعد تثبيت المعاملة. ثم نطلب التخلص من نتائج الإحصاءات القديمة قبل إطلاق الحدث. إذا اختل هذا الترتيب قد يعمل الاتصال اللحظي تماماً ويظهر التنبيه، ومع ذلك تقرأ اللوحة العدد السابق. لذلك أتتبع مصدر الأرقام قبل فتح كود WebSocket.

المعرّف المستخدم للتنبيه يميز الواقعة، أما رقم الطلب فيدل المسؤول على السجل الذي يفتحه. قد نحتاج كليهما: واحد لمنع تكرار الإعلان، وآخر لمعرفة أي طلب يريد المسؤول مراجعته.

ما تقوله

في DashboardCache يوجد مفتاح للإدارة ومفتاح يخص كل زبون. نتيجة شراء واحد تؤثر في الاثنين: ازداد عدد طلبات صاحب الحساب وازداد إجمالي المتجر. لهذا تحذف الدالة ملخص الزبون وتطلب أيضاً حذف مفتاح ملخص المسؤول. هذه ليست عملية تزيد رقماً محفوظاً يدوياً؛ ما نحذفه نسخة مشتقة يمكن للخادم أن يحسبها مجدداً من الطلبات.

إذا سألني الدكتور لماذا نحتاج هذه الخطوة مع أن الواجهة ستعيد الجلب، أرجع إلى مثال العدد عشرة. React يستطيع إعادة إرسال طلب الملخص عند وصول الحدث، لكنه لا يقرر ماذا تعيد Laravel. إذا أعطاه كاش الخادم القديم فسيعرض عشرة مرة ثانية رغم أن الطلب الحادي عشر موجود. أما إذا حذفنا المفتاح هنا ثم طلبت الواجهة الملخص، يستطيع الخادم حساب القيمة الجديدة. كاش الخادم وكاش المتصفح مكانان مختلفان، وكل طرف يعالج النسخة التي يملكها.

بالمقابل، إبطال مفتاح الخادم وحده لا يدفع متصفحاً مفتوحاً إلى القراءة. ستظهر نتيجة الحذف عند الجلب التالي فقط، ولذلك نحتاج إلى إشارة تجعل ذلك الجلب يحصل الآن لا عند التحديث اليدوي القادم.

في صنف OrderCreated نختار قناة إدارة واحدة. هذا مناسب لإعلان الطلبات الجديدة، لأن المسؤول لا يعرف سلفاً أرقام الطلبات التي ستنشأ كي يشترك في كل رقم منها. الحدث يحمل رقم الطلب ومعرّف الواقعة ووقتها فقط؛ لا يحمل صفاً كاملاً جاهزاً للجدول، ولا أسماء الزبائن أو مشترياتهم. عامل البث ينقله عبر Reverb بالاسم الذي سيستمع إليه المتصفح.

ثم أفتح قاعدتي الحماية معاً. في ملف القنوات يجب أن يكون المستخدم الذي عرّفه حارس API مسؤولاً قبل دخول قناة الإدارة. وفي ملف المسارات تقع قائمة جميع الطلبات تحت فحص دور الإدارة أيضاً. لماذا نحتاج الاثنين؟ لأن استقبال خبر عن طلب وقراءة تفاصيل المتجر عمليتان مختلفتان. حتى لو عرف زبون اسم القناة أو عنوان قائمة الإدارة، لا تتحول جلسته العادية إلى جلسة مسؤول. صفحة الملخص نفسها قد تخدم زبوناً ملخصَه الشخصي، أما الأرقام والقائمة الإدارية فتخصان الدور الإداري.

اختيار قناة إدارية واحدة يحل مشكلة الطلب الذي لا تعرفه الواجهة بعد. أما قناة الزبون المرتبطة برقم طلبه فتخدم سؤالاً آخر تماماً: ماذا تغيّر في طلب أعرفه وأملكه بالفعل؟

export function useAdminOrderRealtime() {
  const manager = useRealtimeConnection()
  const queryClient = useQueryClient()
  const isAdmin = useAuthStore((state) => state.isAuthenticated && state.user?.role === ROLES.ADMIN)
  const announcedEvents = useRef(new Set<string>())

  useEffect(() => {
    if (!manager || !isAdmin) return
    let active = true
    const resynchronize = () => {
      const filters: QueryFilters[] = [
        { queryKey: orderKeys.lists() },
        { queryKey: dashboardKeys.summary, exact: true },
      ]
      for (const filter of filters) {
        // Cancel even an initial fetch so a response started before the event cannot win.
        void queryClient.cancelQueries(filter).then(() => {
          const state = useAuthStore.getState()
          if (active && state.isAuthenticated && state.user?.role === ROLES.ADMIN) {
            return queryClient.invalidateQueries(filter)
          }
        })
      }
    }
    const unsubscribe = manager.subscribe({
      channel: "admin.orders",
      event: ".OrderCreated",
      isAllowed: (session) => session.role === ROLES.ADMIN,
      onEvent: (payload) => {
        const orderId = eventOrderId(payload)
        if (orderId === null) return
        resynchronize()
        if (!payload || typeof payload !== "object" || !("event_id" in payload) ||
          typeof payload.event_id !== "string" || payload.event_id.length === 0 ||
          announcedEvents.current.has(payload.event_id)) return
        announcedEvents.current.add(payload.event_id)
        if (announcedEvents.current.size > 100) {
          announcedEvents.current.delete(announcedEvents.current.values().next().value!)
        }
        toastService.info(`طلب جديد #${orderId}`, "يمكنك عرض تفاصيله من صفحة الطلبات")
      },
      onSubscribed: resynchronize,
    })
    return () => { active = false; unsubscribe() }
  }, [manager, queryClient, isAdmin])
}

ما تقوله

في الواجهة رُكّب الخطاف داخل غلاف التطبيق. لهذا يصل الإشعار عندما يقف المسؤول على الملخص، وليس فقط عندما يكون جدول الطلبات مركّباً. داخل الخطاف نقرأ دور الجلسة ولا نركّب الاشتراك الإداري إلا لمسؤول، ثم نتفق مع الخادم على القناة واسم الحدث. هذا الشرط المحلي يمنع اتصالات غير لازمة؛ الحماية التي تمنع غير المخوّل من الاستماع موجودة في Laravel كما رأينا.

عندما تصل الإشارة نفحص رقم الطلب ثم نعيد مزامنة نوعين من البيانات: قوائم الطلبات وملخص اللوحة. نلغي الجلبات القديمة المطابقة حتى لا تصل نتيجة انطلقت قبل الشراء وتعيد العداد إلى عشرة. بعدها نطلب تحديث الاستعلامات المعنية ما دامت الجلسة إدارية. التنبيه عملية منفصلة: نتذكر معرّف الواقعة ضمن مجموعة محدودة حتى لا يُعلَن الطلب نفسه مرتين. وعند نجاح الاشتراك بعد انقطاع نعيد مزامنة البيانات أيضاً؛ نستطيع إصلاح العداد دون أن نزعم أن كل تنبيه فات سيظهر مجدداً.

إذا كان المسؤول يتصفح الإعدادات وقت الشراء، يستطيع تلقي التنبيه من الغلاف المشترك. البيانات غير المعروضة ليس من الضروري تنزيلها كلها فوراً؛ المهم أن يعرف الاستعلام أنها تحتاج تحديثاً عند العودة.

ما تقوله

أختم بما يستطيع المسؤول ملاحظته. في صفحة الملخص تعرض البطاقات بيانات الاستعلام، ومنها عدد الطلبات اليوم والإجمالي. لا توجد عملية تجمع واحداً عند وصول التنبيه. إذا وصل طلبان أثناء انقطاع قصير، لا تحتاج اللوحة إلى تخمين العدد من عدد التنبيهات؛ يكفي أن تقرأ الإجمالي الصحيح من الخادم عند عودة الاتصال. ولهذا نربط شكل البطاقة بمصدر البيانات ونترك البث يحدد متى نحتاج قراءة جديدة.

في صفحة القائمة يتلقى الجدول عناصر الصفحة الحالية التي أعادتها API. عند مشاهدة أول صفحة دون فلتر يُتوقع أن يظهر الطلب الجديد وفق ترتيب الخادم. أما إذا كان المسؤول يعرض الطلبات المُسلَّمة فقط فلن نضيف طلباً معلّقاً إلى تلك القائمة لمجرد أن تنبيهه وصل. قد يتغير الإجمالي ويظهر التنبيه وتظل القائمة المصفاة كما هي؛ هذا تطبيق صحيح للتصفية. هكذا أشرح ثلاثة آثار منفصلة لنفس الحدث: تنبيه فوري، وملخص حديث، وصفوف تحترم الصفحة والفلتر الحاليين.

لذلك أختار في العرض صفحة غير مصفاة ورقم طلب واضحاً. هذا يجعل العلاقة بين الحدث والنتيجة سهلة الملاحظة، ويمنع الخلط بين طلب لم يصل إطلاقاً وطلب استبعده الفلتر بصورة صحيحة.
