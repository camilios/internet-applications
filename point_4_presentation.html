    public function updateStatus(Order $order, string $status): Order
    {
        $updated = DB::transaction(function () use ($order, $status): Order {
            // Route-bound models may be stale after another request changes the order.
            $order = Order::query()->whereKey($order->getKey())->lockForUpdate()->firstOrFail();
            $validTransitions = [
                'pending' => ['confirmed', 'cancelled'],
                'confirmed' => ['processing', 'cancelled'],
                'processing' => ['shipped', 'cancelled'],
                'shipped' => ['delivered'],
                'delivered' => [],
                'cancelled' => [],
            ];
            $currentStatus = $order->status;

            if (! isset($validTransitions[$currentStatus]) ||
                ! in_array($status, $validTransitions[$currentStatus])) {
                throw new \InvalidArgumentException(
                    "Cannot transition from '{$currentStatus}' to '{$status}'."
                );
            }

            if ($status === 'cancelled' && $order->reservation_id !== null) {
                if (! $this->inventoryClient->releaseStock($order->reservation_id)) {
                    throw new InventoryUnavailableException('Stock release could not be confirmed.');
                }
            }

            $order->update(['status' => $status]);
            $this->notifyAfterCommit($order->user, $order, 'order_status_updated');
            $this->broadcastAfterCommit($order, new OrderStatusUpdated((string) Str::uuid(), $order->id, $status, now()->toISOString()));

            return $order->fresh('items.product');
        });

        return $updated;
    }

    private function broadcastAfterCommit(Order $order, OrderCreated|OrderStatusUpdated $event): void
    {
        DB::afterCommit(function () use ($order, $event): void {
            try {
                DashboardCache::forgetCache($order->user_id);
            } catch (\Throwable $exception) {
                Log::error('Order dashboard cache invalidation failed', ['order_id' => $order->id, 'error_type' => $exception::class]);
            }
            try {
                event($event);
            } catch (\Throwable $exception) {
                // A committed order must not be compensated due to a secondary broadcast failure.
                Log::error('Order broadcast enqueue failed after commit', ['order_id' => $order->id, 'error_type' => $exception::class]);
            }
        });
    }

ما تقوله

لنفترض أن الزبون يشاهد الطلب ٤٢ بحالة معلّق، وأن المسؤول وافق عليه من متصفح آخر. في updateStatus نعيد قراءة صف الطلب ونقفله أثناء الانتقال. هذه الخطوة تمنع موافقتين مبنيتين على حالة قديمة من إنتاج روايتين متناقضتين. إذا كان الانتقال صالحاً تُكتب الحالة الجديدة، ثم ننشئ حدثاً يصف رقم الطلب والحالة ووقت التغيير ومعرّف الواقعة. ما يعنينا في هذه الدالة هو تتابع الحفظ ثم التبليغ، لا بقية حالات الانتقال أو معالجة المخزون.

في broadcastAfterCommit نؤخر إطلاق الحدث إلى ما بعد تثبيت المعاملة. لو أبلغنا الصفحة قبل الحفظ، فقد تعيد الجلب وتحصل على حالة معلّق مرة أخرى. والأسوأ أن يرى الزبون موافقة ثم تتراجع عنها قاعدة البيانات بسبب فشل المعاملة. الآن يصبح معنى الإشارة واضحاً: يوجد تحديث محفوظ لهذا الطلب، ويمكن للواجهة أن تسأل عنه بثقة. لا تعيد Laravel رسم الصفحة بنفسها، بل تبلغ مشتركاً يستطيع إعادة رسمها.

في تجربتي أترك صفحة الزبون مفتوحة قبل الضغط على الموافقة. إن لم تتغير أسأل أولاً هل صار الطلب مؤكداً فعلاً في قاعدة البيانات، ثم أتتبع وصول الإشارة؛ التشخيصان مختلفان.

ما تقوله

السؤال التالي: أي متصفح يسمع الإشارة؟ في صنف الحدث أبدأ من اسم القناة. الاسم يبنى من رقم الطلب، فيذهب تحديث ٤٢ إلى قناة ٤٢، لا إلى قناة عامة لكل زبائن المتجر. اسم الحدث معروف للواجهة، والحمولة تكفي لتحديد ما تغير من دون تضمين بريد الزبون أو كل بنود الشراء. البث يمر بطابور البث الخاص به؛ عامل البريد ليس حلقة ضمن مسار عرض الحالة.

وجود رقم في اسم القناة لا يثبت الخصوصية بمفرده. لهذا أفتح قاعدة التفويض: يدخل إليها المستخدم المصادق عليه ورقم الطلب المطلوب، ثم تتحقق من وجود صف يجمع هذا الرقم ومعرّف ذلك المستخدم. لو طلب زبون آخر سماع القناة ذاتها، يبقى رقم الطلب صحيحاً ولكن شرط الملكية يفشل. فحص الواجهة يحسن السلوك المعتاد، أما هذا القرار على خادم Laravel فهو ما يمنع الاشتراك نفسه.

المعلومة التي تعرضها الشارة يمكن جلبها أيضاً عبر HTTP، لكن حق قراءة تفاصيل الطلب عبر ذلك المسار لا يحول القناة تلقائياً إلى قناة مسموحة. لكل قناة طلب تفويض خاص بها.

export function useOrderStatusRealtime(order: Pick<Order, "id" | "user_id"> | undefined) {
  const manager = useRealtimeConnection()
  const queryClient = useQueryClient()
  const userId = useAuthStore((state) => state.isAuthenticated ? state.user?.id : undefined)
  const orderId = order?.id
  const ownerId = order?.user_id

  useEffect(() => {
    if (!manager || !orderId || !Number.isInteger(orderId) || ownerId !== userId || !userId) return
    let active = true
    const filters: QueryFilters[] = [
      { queryKey: orderKeys.detail(orderId), exact: true },
      { queryKey: orderKeys.lists() },
    ]
    const resynchronize = () => {
      for (const filter of filters) {
        void queryClient.cancelQueries(filter).then(() => {
          const state = useAuthStore.getState()
          if (active && state.isAuthenticated && state.user?.id === ownerId) {
            return queryClient.invalidateQueries(filter)
          }
        })
      }
    }
    const unsubscribe = manager.subscribe({
      channel: `orders.${orderId}`,
      event: ".OrderStatusUpdated",
      isAllowed: (session) => session.userId === ownerId,
      onEvent: (payload) => {
        if (eventOrderId(payload) !== orderId) return
        resynchronize()
        announceOrderStatus(manager, payload, orderId, userId)
      },
      onSubscribed: resynchronize,
    })
    return () => { active = false; unsubscribe() }
  }, [manager, queryClient, userId, orderId, ownerId])
}

ما تقوله

في صفحة التفاصيل أبدأ من العلاقة بين الاستعلام والخطاف والشارة. الصفحة تجلب الطلب أول مرة لتعرض بياناته حتى لو لم يصل أي بث. بعد وصول البيانات يمرّر المكوّن الطلب إلى خطاف الاستماع، وفي العرض تُشتق تسمية الحالة من بيانات الاستعلام نفسه. لهذا لا نحتاج إلى نسخة ثانية من الحالة في متغير مستقل قد يختلف مع الخادم.

في الخطاف نتحقق من المالك ونستمع إلى القناة الخاصة برقم الطلب وإلى اسم الحدث الذي قرأناه في الخلفية. حين تصل إشارة مطابقة نلغي الجلب السابق ثم نعلّم بيانات التفاصيل والقائمة أنها بحاجة إلى تحديث، فتجلب الاستعلامات المفتوحة القيم الحالية. الغرض من إلغاء الجلب القديم أن رداً انطلق قبل الموافقة قد يصل متأخراً. وعند نجاح الاشتراك نعيد المزامنة أيضاً؛ لو حصلت الموافقة أثناء انقطاع الشبكة فلا نعتمد على أن يعيد WebSocket تشغيل التاريخ الذي فاتنا. النتيجة هي تحديث البيانات التي ترسم الشارة، لا طلب إعادة تحميل الصفحة.

يمكن اختبار الفكرة ذهنياً بإغلاق الاتصال للحظة: لن تصل إشارة الموافقة في وقتها، لكن أول إعادة مزامنة ناجحة تستطيع إظهار القيمة الحالية. ما نحتاجه للشارة هو أحدث حالة، وليس إعادة كل حالة وسيطة.

export function useOrderListStatusRealtime(orders: readonly Pick<Order, "id" | "user_id">[] | undefined) {
  const manager = useRealtimeConnection()
  const queryClient = useQueryClient()
  const userId = useAuthStore((state) => state.isAuthenticated && state.user?.role === ROLES.CUSTOMER ? state.user.id : undefined)
  // A REST refetch returns new objects. Keep subscriptions stable when the visible IDs are unchanged.
  const orderIds = [...new Set((orders ?? [])
    .filter((order) => order.user_id === userId && Number.isInteger(order.id) && order.id > 0)
    .map((order) => order.id))].sort((left, right) => left - right).join(",")

  useEffect(() => {
    if (!manager || !userId || !orderIds) return
    let active = true
    let timer: ReturnType<typeof setTimeout> | undefined
    const pendingIds = new Set<number>()
    const resynchronize = (orderId: number) => {
      pendingIds.add(orderId)
      if (timer !== undefined) return
      // Coalesce the page's subscription-success/event burst into one list refetch.
      timer = setTimeout(() => {
        timer = undefined
        const state = useAuthStore.getState()
        if (!active || !state.isAuthenticated || state.user?.id !== userId || state.user.role !== ROLES.CUSTOMER) {
          pendingIds.clear()
          return
        }
        const filters: QueryFilters[] = [
          { queryKey: orderKeys.lists() },
          ...[...pendingIds].map((id) => ({ queryKey: orderKeys.detail(id), exact: true })),
        ]
        pendingIds.clear()
        for (const filter of filters) {
          void queryClient.cancelQueries(filter).then(() => {
            const state = useAuthStore.getState()
            if (active && state.isAuthenticated && state.user?.id === userId && state.user.role === ROLES.CUSTOMER) {
              return queryClient.invalidateQueries(filter)
            }
          })
        }
      }, 20)
    }
    const unsubscribe = orderIds.split(",").map(Number).map((orderId) => manager.subscribe({
      channel: `orders.${orderId}`,
      event: ".OrderStatusUpdated",
      isAllowed: (session) => session.userId === userId && session.role === ROLES.CUSTOMER,
      onEvent: (payload) => {
        if (eventOrderId(payload) !== orderId) return
        resynchronize(orderId)
        announceOrderStatus(manager, payload, orderId, userId)
      },
      onSubscribed: () => resynchronize(orderId),
    }))
    return () => {
      active = false
      if (timer !== undefined) clearTimeout(timer)
      pendingIds.clear()
      for (const release of unsubscribe) release()
    }
  }, [manager, queryClient, userId, orderIds])
}

ما تقوله

الآن أرجع للسيناريو الذي كان يسبب اللبس: الزبون يقف على قائمة طلباته، ولم يفتح صفحة التفاصيل أصلاً. لو اكتفينا بالخطاف السابق فلن يوجد في هذه الشاشة مستمع يعمل نيابة عنها. لهذا تربط صفحة القائمة بيانات الصفحة الحالية بخطاف مستقل، في حين يعرض كل صف تسمية الحالة الآتية من بيانات ذلك الصف. إذا تغيّرت نتيجة الاستعلام يتغير الصف من تلقاء نفسه.

في خطاف القائمة ننتقي الطلبات المملوكة للمستخدم والظاهرة حالياً، ونحوّلها إلى مجموعة أرقام مستقرة. عند تبدّل كائنات React بسبب جلب جديد مع بقاء نفس الأرقام لا نريد فك الاشتراك وتركيبه من الصفر بلا سبب. ينشأ مستمع لكل طلب ظاهر، وإذا وصلت عدة إشارات متقاربة تُجمع خلال نافذة قصيرة ثم تحدَّث القائمة والتفاصيل المتأثرة. لذلك قد تختفي بطاقة من قائمة تعرض المعلّق فقط بعد أن يصبح الطلب مؤكداً. هذا اختفاء صحيح تفرضه التصفية، وليس فقداناً للتحديث. وعند مغادرة الصفحة يحرر الخطاف اشتراكاتها.

هذه النقطة تفسر اختلافاً عملياً بين الشاشتين: التفاصيل تعرف رقماً واحداً طوال وجودها، أما القائمة فمجموعة أرقامها تتبدل عند الانتقال بين الصفحات أو تغيير الفلتر. لذلك يختلف إعداد المستمعين مع أن الحدث واحد.

export function announceOrderStatus(
  manager: RealtimeConnectionManager,
  payload: unknown,
  orderId: number,
  ownerId: number,
) {
  const state = useAuthStore.getState()
  if (!state.isAuthenticated || !state.token || state.user?.role !== ROLES.CUSTOMER || state.user.id !== ownerId ||
    !payload || typeof payload !== "object" || !("order_id" in payload) || payload.order_id !== orderId ||
    !("event_id" in payload) || typeof payload.event_id !== "string" || !eventIdPattern.test(payload.event_id) ||
    !("status" in payload) || typeof payload.status !== "string" || !Object.hasOwn(ORDER_STATUS_LABELS, payload.status)) return

  const generation = manager.getSessionGeneration()
  let delivery = announced.get(manager)
  if (!delivery || delivery.generation !== generation) {
    delivery = { generation, eventIds: new Set() }
    announced.set(manager, delivery)
  }
  if (delivery.eventIds.has(payload.event_id)) return
  delivery.eventIds.add(payload.event_id)
  if (delivery.eventIds.size > 100) delivery.eventIds.delete(delivery.eventIds.values().next().value!)

  toastService.info(
    `تم تحديث حالة الطلب #${orderId}`,
    `الحالة الجديدة: ${ORDER_STATUS_LABELS[payload.status as OrderStatus]}`,
  )
}

ما تقوله

أخيراً أميّز بين ما أراه في الشارة وما أراه في التنبيه. الشارة تعكس أحدث بيانات الطلب التي جلبتها الصفحة، أما التنبيه فيصف واقعة التغيير لحظة وصولها. في الصفحتين نستدعي دالة الإعلان نفسها. قبل عرض شيء تتأكد أن المستخدم الحالي هو الزبون المالك وأن الرقم والحالة ومعرّف الحدث صالحون. لذلك لن تعرض نصاً مجهولاً أو تنبيهاً لطلب آخر لمجرد أن هناك اتصالاً مفتوحاً.

ممّا قد يحدث أن يصل الحدث نفسه لأكثر من مستمع، أو أن ينتقل الزبون بين القائمة والتفاصيل أثناء المعالجة. تحتفظ الدالة بمجموعة محدودة من معرّفات الإعلانات الحديثة للجلسة؛ إذا رأت المعرّف نفسه لا تكرر التنبيه، أما تحديث آخر لنفس الطلب فله معرّف جديد. ما ننتظره أمام الدكتور هو أن تتغير الحالة في الشاشة المفتوحة دون تحديث يدوي، ويظهر تنبيه واحد للتغيير الجديد، ويبقى مستخدم الحساب الآخر خارج القناة.

لو انتقلت إلى التفاصيل بعد حدوث الموافقة فلا يلزم أن يظهر تنبيه جديد لمجرد أنك فتحتها. الذي ينبغي أن تجده هناك هو الحالة الصحيحة؛ التنبيه مخصص لواقعة وصلت أثناء المتابعة.
