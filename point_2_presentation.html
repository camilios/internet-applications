    public function handle(MessageBroker $broker, EmailMessageProcessor $processor, MailDeliveryConfiguration $mailConfiguration): int
    {
        try {
            // Check before connecting to the database or consuming any message.
            $mode = $mailConfiguration->assertDeliverable();
        } catch (MailDeliveryConfigurationException $exception) {
            $this->error($exception->getMessage());

            return self::FAILURE;
        }
        if ($mode === 'mock') {
            $this->warn('MOCK DELIVERY enabled: order confirmations will not send external email.');
            Log::warning('MOCK DELIVERY enabled for order email worker; no external email will be sent');
        }
        DB::connection()->useWriteConnectionWhenReading();
        do {
            try {
                $message = $broker->receive();
                if ($message !== null) {
                    $processor->handle($message);
                } elseif (! $this->option('once')) {
                    sleep(1);
                }
            } catch (MailDeliveryConfigurationException $exception) {
                $broker->close();
                $this->error($exception->getMessage());

                return self::FAILURE;
            } catch (\Throwable $exception) {
                // Closing leaves the delivery unacknowledged; RabbitMQ redelivers on recovery.
                $broker->close();
                Log::error('Order email worker recovering', ['error_type' => $exception::class]);
                if ($this->option('once')) {
                    return self::FAILURE;
                }
                sleep(config('rabbitmq.reconnect_delay'));
            }
        } while (! $this->option('once'));
        $broker->close();

        return self::SUCCESS;
    }

ما تقوله

أبدأ من تعريف عامل البريد في Compose. قد ينشئ زبون طلباً ثم يغلق الصفحة، ومع ذلك يستمر تجهيز التأكيد. السبب أن الإرسال يحدث داخل حاوية لها أمر تشغيل خاص، وليس في المتصفح ولا في آخر سطر من متحكم الطلب. صحيح أن العامل يستعمل صورة Laravel وإعدادات المشروع، لكنه عملية منفصلة: الناشر يضع حدث الشراء عند الوسيط، وهذا الأمر يأخذه ليحوّله إلى بريد.

في بداية handle يفحص العامل إعداد البريد قبل أن يسحب شيئاً من الطابور. لو كان الإعداد غير صالح يخرج بخطأ، ولو سُمِح بوضع تجريبي يعلن ذلك بوضوح. بعدها يقرأ من قاعدة البيانات الرئيسية ويسحب الرسائل في حلقة. أهمية القراءة من الرئيسية أن الحدث قد يصل بعد الشراء مباشرة؛ لا نريد أن يظن العامل أن الطلب مفقود لمجرد أن نسخة القراءة لم تتحدث بعد. وعند فراغ الطابور ينتظر، ولا تُحسب هذه الحالة كفشل إرسال.

مثال ذلك أن خادم الطلبات قد يستقبل ثلاثة طلبات متتالية بينما ينتظر العامل رد مزود البريد على الرسالة الأولى. لا ينتظر أصحاب الطلبين الآخرين ذلك الرد لكي يحصلوا على رقمي طلبيهما.

    public function receive(): ?IncomingMessage
    {
        // Polling keeps heartbeat/socket handling bounded, including on an idle queue.
        $message = $this->channel()->basic_get(config('rabbitmq.queue'), false);
        if ($message === null) {
            return null;
        }
        $headers = $message->has('application_headers') ? $message->get('application_headers')->getNativeData() : [];
        $attempt = filter_var($headers['attempt'] ?? 1, FILTER_VALIDATE_INT);

        return new IncomingMessage(
            $message->getBody(),
            $attempt === false ? 1 : max(1, $attempt),
            fn () => $message->ack(),
        );
    }

    public function handle(IncomingMessage $message): void
    {
        $event = null;
        try {
            $event = $this->decode($message->body);
            Log::info('Order email processing', ['event_id' => $event['event_id'], 'attempt' => $message->attempt]);
            $this->sender->send($event);
        } catch (MailDeliveryConfigurationException $exception) {
            // Misconfiguration is operational, not a failed delivery attempt. Preserve the original message.
            throw $exception;
        } catch (\Throwable $exception) {
            $permanent = $exception instanceof PermanentMessageException || $exception instanceof JsonException;
            $exhausted = $message->attempt >= config('rabbitmq.max_attempts');
            $eventId = $event['event_id'] ?? (string) Str::uuid();
            if ($permanent || $exhausted) {
                // Confirm the DLQ publication before acknowledging the original delivery.
                $this->broker->publish($message->body, $eventId, config('rabbitmq.dead_letter_exchange'), config('rabbitmq.dead_letter_routing_key'), $message->attempt);
            } else {
                $retryIndex = min($message->attempt, count(config('rabbitmq.retry_delays')));
                $this->broker->publish($message->body, $eventId, config('rabbitmq.retry_exchange'), 'retry.'.$retryIndex, $message->attempt + 1);
            }
            Log::warning('Order email deferred', [
                'event_id' => $eventId, 'attempt' => $message->attempt,
                'destination' => $permanent || $exhausted ? 'dead_letter' : 'retry', 'error_type' => $exception::class,
            ]);
            $message->acknowledge();

            return;
        }

        Log::info('Order email processed', ['event_id' => $event['event_id'], 'attempt' => $message->attempt]);
        $message->acknowledge();
    }

ما تقوله

في receive أركّز على basic_get. المعامل false هنا قيمة لخيار معناه «بدون إقرار»، لذلك نتيجته الفعلية أن الإقرار التلقائي معطّل. لا تختفي الرسالة من مسؤولية الوسيط بمجرد أخذها؛ يعطينا الكود جسمها ورقم المحاولة وطريقة لإقرارها لاحقاً. إن لم يجد رسالة أصلاً، تعود نتيجة فارغة ولا ينشئ العامل بريداً من لا شيء.

بعدها أنظر في المعالج إلى بداية مسار النجاح ونهايته معاً. يفهم محتوى الحدث ويتحقق أنه يمثل إنشاء طلب، ثم يمرره إلى خدمة الإرسال. فقط بعد رجوع تلك الخدمة بنجاح يؤكد للوسيط أنه انتهى. لهذا لا أستدل على إرسال البريد من عبارة «استلم العامل الحدث». الاستلام يثبت أن الطابور أوصل العمل؛ أما التنفيذ فيحتاج عنوان الزبون وناقل بريد فعلياً، وهما ما سنتتبعه الآن.

لو وصلت رسالة فيها رقم طلب فقط، ولم نستطع التحقق من أنها حدث الإنشاء المتوقع، فلا يكفي أن تكون موجودة في الطابور لنرسل شيئاً. الوسيط ينقل عملاً، أما صحة العمل التجارية فيتحقق منها التطبيق.

    private function validateMailer(string $name, array $ancestors): string
    {
        if ($name === '' || in_array($name, $ancestors, true)) {
            throw new MailDeliveryConfigurationException('Order email mailer configuration is missing or circular.');
        }
        $mailer = config('mail.mailers.'.$name);
        if (! is_array($mailer)) {
            throw new MailDeliveryConfigurationException('Order email mailer is not configured.');
        }
        if (isset($mailer['url'])) {
            try {
                // Mirror Laravel MailManager so a MAIL_URL override cannot hide a non-delivery transport.
                $mailer = (new ConfigurationUrlParser)->parseConfiguration($mailer);
                $mailer['transport'] = $mailer['driver'] ?? null;
            } catch (\Throwable) {
                throw new MailDeliveryConfigurationException('Order email mailer URL is invalid.');
            }
        }
        $transport = $mailer['transport'] ?? null;
        $explicitMock = config('mail.order_email_allow_mock') && app()->environment('local', 'testing');
        if (in_array($transport, ['array', 'log'], true) && $ancestors === []
            && ($explicitMock || ($transport === 'array' && app()->environment('testing')))) {
            return 'mock';
        }
        if (in_array($transport, ['failover', 'roundrobin'], true)) {
            $children = $mailer['mailers'] ?? [];
            if (! is_array($children) || $children === []) {
                throw new MailDeliveryConfigurationException('Order email composite mailer has no delivery transports.');
            }
            foreach ($children as $child) {
                if (! is_string($child)) {
                    throw new MailDeliveryConfigurationException('Order email composite mailer configuration is invalid.');
                }
                $this->validateMailer($child, [...$ancestors, $name]);
            }

            return 'external';
        }
        if (! in_array($transport, ['smtp', 'sendmail', 'mail', 'mailgun', 'ses', 'ses-v2', 'postmark', 'resend'], true)) {
            throw new MailDeliveryConfigurationException('Order emails require a delivery mailer. Explicit log/array mock mode is allowed only in local/testing environments; non-delivery fallbacks are not allowed.');
        }
        if ($transport === 'smtp') {
            $host = $mailer['host'] ?? '';
            $validHost = is_string($host) && (filter_var($host, FILTER_VALIDATE_DOMAIN, FILTER_FLAG_HOSTNAME) !== false || filter_var($host, FILTER_VALIDATE_IP) !== false);
            $validPort = filter_var($mailer['port'] ?? 25, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1, 'max_range' => 65535]]) !== false;
            if (! $validHost || ! $validPort) {
                throw new MailDeliveryConfigurationException('Configure the external SMTP provider host and port before starting the order email worker.');
            }
        }

        return 'external';
    }

ما تقوله

في إعداد SMTP توجد ثلاث معلومات مختلفة من المفيد تمييزها: الناقل الذي ستستعمله Laravel، عنوان الخادم الخارجي، وبيانات الاعتماد التي تأتي من البيئة. في تجربة Resend يشير المضيف إلى خدمة Resend، ولا تُنسخ كلمة الاعتماد إلى كود الطلب أو إلى محتوى الرسالة. إذا كان الناقل المختار يسجل النص محلياً فلن يصل شيء إلى صندوق العميل حتى لو كان قالب البريد صحيحاً تماماً.

لهذا نفحص طريقة التسليم في الدالة الثانية. يسمح المشروع في حالات محلية أو اختبارية محددة بناقل تجريبي، وتعيد الدالة حينها وصفاً واضحاً بأنه mock. أما SMTP فيحتاج إعداد مضيف ومنفذ صالحين حتى يُعد ناقلاً خارجياً. كلمة external هنا تصف المسار المختار، لا نتيجة إرسال هذه الرسالة بعينها؛ فحص صيغة العنوان لا يثبت أن بيانات الدخول قُبلت أو أن مزوّد البريد يعمل الآن. البرهان التالي موجود في استدعاء الإرسال نفسه.

يمكن إذن أن ينجح اختبار قالب الرسالة باستخدام ناقل محلي وأن يفشل إرسال حقيقي لنفس القالب بسبب إعداد المزوّد. هذا لا يناقض الاختبار؛ كل واحد منهما يجيب عن سؤال مختلف.

    public function send(array $event): void
    {
        $mode = $this->mailConfiguration->assertDeliverable();

        $outbox = OutboxEvent::query()->useWritePdo()->find($event['event_id']);
        if ($outbox === null || $outbox->event_type !== 'OrderCreated' || $outbox->order_id !== $event['order_id']) {
            throw new PermanentMessageException('Message does not identify a recorded order event.');
        }

        // Unique row + row lock serializes duplicate deliveries, even across worker processes.
        DB::table('notification_deliveries')->insertOrIgnore([
            'event_id' => $event['event_id'], 'order_id' => $event['order_id'],
            'created_at' => now(), 'updated_at' => now(),
        ]);
        DB::transaction(function () use ($event, $mode): void {
            $delivery = NotificationDelivery::query()->whereKey($event['event_id'])->lockForUpdate()->firstOrFail();
            if ($delivery->delivered_at !== null) {
                // A provider change does not automatically replay earlier mock or unclassified deliveries.
                return;
            }

            $order = Order::query()->useWritePdo()->with(['user', 'items.product'])->find($event['order_id']);
            if ($order === null || $order->user === null || ! filter_var($order->user->email, FILTER_VALIDATE_EMAIL)) {
                throw new PermanentMessageException('Order recipient is unavailable.');
            }

            $sent = Mail::to($order->user->email)->send(new OrderConfirmation($order, $event['event_id']));
            if ($sent === null) {
                throw new RuntimeException('Mail delivery was cancelled before sending.');
            }
            // SMTP cannot participate in this transaction. A crash here can resend; stable Message-ID helps providers deduplicate.
            $delivery->update(['delivered_at' => now(), 'delivery_mode' => $mode]);
            Log::info($mode === 'mock' ? 'MOCK DELIVERY: order confirmation processed without an external email' : 'Order confirmation accepted by delivery transport', [
                'event_id' => $event['event_id'], 'delivery_mode' => $mode,
                'transport_message_id' => substr(preg_replace('/[^A-Za-z0-9@._:+\/-]/', '', $sent->getMessageId()), 0, 255),
            ]);
        }, 1);
    }

ما تقوله

الجزء المهم في send يبدأ بالربط بين معرّف الحدث والطلب المسجّل أصلاً. لا نثق بعنوان مستلم قادم من حمولة خارجية؛ نتحقق من ارتباط الحدث بطلب حقيقي، ثم نحضر الطلب وصاحبه وعناصره من قاعدة المتجر. بذلك يكون البريد موجهاً إلى صاحب الطلب، وتكون التفاصيل المستخدمة في النص هي التفاصيل التي حفظها المتجر.

في قلب المعاملة نأخذ قفلاً على سجل التسليم الخاص بالحدث. لو استيقظ عاملان للحدث نفسه، لا ينبغي أن يقررا الإرسال في اللحظة نفسها. إذا كان وقت التسليم محفوظاً يرجع العامل دون إنشاء بريد جديد. وفي أول معالجة يمرّر عنوان صاحب الطلب ورسالة التأكيد إلى Mail، ثم يحفظ الوقت ووضع التسليم بعد النجاح. توجد حدود واضحة لهذا الترتيب: إذا قبل المزوّد الرسالة ثم تعطّل العامل قبل حفظ الوقت، لا تستطيع معاملة MySQL سحب البريد من المزوّد. لذلك نخفّف التكرار ولا نعد بأنه مستحيل.

ولهذا نحتفظ أيضاً بوضع التسليم مع وقته. عند المراجعة نستطيع التفريق بين طلب عولج في تجربة محلية وطلب سُلّم لناقل خارجي، حتى لو كان رقم الطلب ظاهراً في سجل الحالتين.

class OrderConfirmation extends Mailable
{
    public function __construct(public readonly Order $order, private readonly string $eventId) {}

    public function envelope(): Envelope
    {
        return new Envelope(subject: 'Order #'.$this->order->id.' confirmation');
    }

    public function headers(): Headers
    {
        $domain = parse_url((string) config('app.url'), PHP_URL_HOST) ?: 'localhost';

        return new Headers(
            messageId: $this->eventId.'@'.$domain,
            text: $this->usesResendSmtp() ? ['Resend-Idempotency-Key' => 'order-confirmation/'.$this->eventId] : [],
        );
    }

    private function usesResendSmtp(): bool
    {
        $mailer = config('mail.mailers.'.($this->mailer ?? config('mail.default')), []);
        if (! is_array($mailer)) {
            return false;
        }
        if (isset($mailer['url'])) {
            try {
                // Laravel allows MAIL_URL to override both the host and transport.
                $mailer = (new ConfigurationUrlParser)->parseConfiguration($mailer);
                $mailer['transport'] = $mailer['driver'] ?? null;
            } catch (\Throwable) {
                return false;
            }
        }

        return ($mailer['transport'] ?? null) === 'smtp'
            && is_string($mailer['host'] ?? null)
            && strtolower(rtrim($mailer['host'], '.')) === 'smtp.resend.com';
    }

    public function content(): Content
    {
        return new Content(view: 'mail.order-confirmation');
    }
}

ما تقوله

أنهي التتبع بما يستطيع الزبون رؤيته. في صنف الرسالة يُبنى الموضوع من رقم الطلب، ويُستخدم معرّف ثابت للرسالة. وعندما يتأكد الكود أن النقل هو SMTP عبر Resend، يضيف ترويسة بمفتاح مشتق من الحدث نفسه كي يستطيع المزوّد تمييز إعادة المحاولة. ليس المقصود أن يصبح كل بريد يطلبه الزبون مطابقاً لبريد سابق؛ طلب جديد له حدث مختلف، أما إعادة معالجة الحدث نفسه فتحتفظ بهويته.

في القالب أشير إلى رقم الطلب وبنوده ومجموعه. هذه هي الصلة بين بيانات العامل والنتيجة خارج التطبيق: أستطيع مطابقة موضوع الرسالة ومجموعها مع الطلب الذي أنشأته. إذا سألني الدكتور ماذا يثبت السجل، أفرق بين ثلاث مراحل: تنفيذ العامل، وقبول ناقل خارجي للرسالة، ثم وصول نسخة مطابقة إلى صندوق الاختبار. ظهور رقم الطلب الصحيح داخل الصندوق هو ما يثبت المرحلة الأخيرة؛ تشغيل الحاوية أو تسجيل نتيجة تجريبية لا يثبتها.

في العرض أكتفي بذكر عنوان الرسالة ورقم الطلب، لا بيانات اعتماد المزوّد. وإذا تأخر ظهورها في الصندوق أرجع إلى سجل نفس الحدث أولاً بدل أن أنشئ عدة مشتريات وأخلط بين تأكيداتها.
