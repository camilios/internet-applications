export function createEchoClient(session: RealtimeSession): RealtimeClient | null {
  const key = import.meta.env.VITE_REVERB_APP_KEY
  if (!key) return null

  let active = true
  const requests = new Set<AbortController>()
  const secure = (import.meta.env.VITE_REVERB_SCHEME ?? window.location.protocol.replace(":", "")) === "https"
  const port = Number(import.meta.env.VITE_REVERB_PORT ?? (secure ? 443 : 8080))
  const echo = new Echo({
    broadcaster: "reverb",
    withoutInterceptors: true,
    Pusher,
    key,
    wsHost: import.meta.env.VITE_REVERB_HOST || window.location.hostname,
    wsPort: port,
    wssPort: port,
    forceTLS: secure,
    enabledTransports: ["ws", "wss"],
    // Preserve the existing Axios token, timeout and expired-session handling.
    authorizer: (channel: { name: string }) => ({
      authorize: (socketId: string, callback: (error: Error | null, data: ChannelAuthorization | null) => void) => {
        if (!active || !isCurrentSession(session)) {
          callback(new Error("Authentication session changed"), null)
          return
        }
        const controller = new AbortController()
        requests.add(controller)
        void apiClient.post<ChannelAuthorization>("/broadcasting/auth", {
          socket_id: socketId,
          channel_name: channel.name,
        }, { signal: controller.signal }).then(({ data }) => {
          if (active && isCurrentSession(session)) callback(null, data)
          else callback(new Error("Authentication session changed"), null)
        }).catch(() => {
          callback(new Error("Private channel authorization failed"), null)
        }).finally(() => requests.delete(controller))
      },
    }),
  })

  return {
    private: (name) => {
      const channel = echo.private(name)
      return {
        listen: (event, listener) => { channel.listen(event, listener) },
        stopListening: (event, listener) => { channel.stopListening(event, listener) },
        onSubscribed: (listener) => {
          // Bind directly so the exact subscription callback can be removed on cleanup.
          channel.subscription.bind("pusher:subscription_succeeded", listener)
          return () => channel.subscription.unbind("pusher:subscription_succeeded", listener)
        },
      }
    },
    leave: (name) => { echo.leave(name) },
    disconnect: () => {
      active = false
      for (const request of requests) request.abort()
      requests.clear()
      echo.disconnect()
    },
  }
}

ما تقوله

أبدأ من متصفح زبون يريد سماع تحديثات الطلب ٤٢. هناك خطوتان لا أخلط بينهما: أن يفتح اتصال شبكة مع خادم البث، وأن يحصل على إذن لسماع قناة خاصة. في createEchoClient أبدأ من استدعاء القناة الخاصة ثم أنتقل إلى دالة التفويض المميزة في نفس الكود. عندما تطلب المكتبة الاشتراك، ترسل الواجهة طلباً منفصلاً إلى Laravel يحمل اسم القناة ورقم اتصال WebSocket الحالي. لا نرسل حقل دور يختاره الزبون ونطلب من الخادم تصديقه.

بعد رد الخادم لا تمرر الواجهة بيانات التفويض إلى المكتبة إلا إذا ظلت الجلسة نفسها نشطة. لو خرج المستخدم أثناء الطلب ثم وصلت نتيجة مسموحة بعد تبديل الحساب، لا يجوز تطبيق إذن الحساب الأول على اتصال الحساب الثاني. ولو رفض الخادم، تُبلَّغ المكتبة بالفشل بلا بيانات اشتراك. لهذا قد ترى اتصالاً ناجحاً في المتصفح وفي الوقت نفسه لا تصل رسالة من القناة الخاصة: فتح الاتصال لا يساوي قبول الاشتراك.

لتحديد موضع الخلل يكفيني أن أسأل أي مرحلة نجحت. إن كان الاتصال مفتوحاً ولكن تفويض القناة مرفوضاً، فلا يفيد أن أغيّر المكوّن الذي يرسم قائمة الطلبات؛ المشكلة أسبق من رسم البيانات.

ما تقوله

من أين عرف Laravel من يسأل؟ نفس عميل Axios الذي يستخدمه التطبيق يقرأ رمز الجلسة الحالي عند إرسال الطلب ويضعه في ترويسة Bearer. هذا الرمز يخص هوية المستخدم، وهو غير رقم اتصال WebSocket وغير الإذن الخاص بقناة واحدة. وجود اسم القناة ورقم الاتصال بلا رمز صالح لا يعرّف أحداً لخادم التطبيق.

في إعداد المصادقة يستخدم حارس API نظام JWT ومزوّد المستخدمين. ثم في ملف المسارات نلاحظ أن نقطة تفويض البث موجودة داخل مجموعة auth:api. هكذا يصل الطلب أولاً إلى فحص الهوية، ثم إلى متحكم البث الذي يطبّق قاعدة القناة. حد الطلبات الملحق بالمسار يحد عدد المحاولات، لكنه لا يقرر إن كان الطلب ملكاً للمستخدم. إذا لم توجد هوية صالحة فالمتوقع 401. أما صاحب الرمز الصالح فقد يُرفض لاحقاً بسبب طلب قناة ليست له؛ في هذه الحالة المشكلة في الصلاحية لا في تسجيل الدخول.

إذاً رقم الاتصال يحدد أين سيُستخدم الإذن، والرمز يثبت من يطلبه، والقاعدة تختار ما يستطيع سماعه. وجود قيمة صحيحة لإحدى الثلاث لا يعوّض غياب القيمتين الأخريين.

ما تقوله

الآن أقرأ قاعدتي القنوات باعتبارهما جوابين مختلفين عن السؤال نفسه. لقناة الإدارة السؤال هو: هل المستخدم الذي عرّفه حارس API يحمل دور المسؤول؟ لا يهم أي طلب يملكه هذا المستخدم شخصياً، لأن القناة تبث أخباراً عن المتجر كله. الزبون العادي قد يكون مسجلاً دخوله بصورة صحيحة ومع ذلك ليس من حقه استقبال هذا التدفق.

أما لقناة طلب بعينه فالسؤال ليس «هل هو مسؤول؟» بل «هل يوجد طلب بهذا الرقم يملكه هذا المستخدم؟». الرقم يأتي من اسم القناة، أما هوية المالك المحتمل فتأتي من المستخدم المصادق عليه. يفحص الكود صيغة الرقم ويقرأ من الاتصال الرئيسي؛ وهذا مهم عندما ينشئ الزبون طلباً ثم يشترك فيه مباشرة. البحث يجمع رقم الطلب ومعرّف المستخدم في الاستعلام ذاته. لا فرق من جهة قرار الاشتراك بين طلب غير موجود وطلب لزبون آخر. كذلك لا يوجد استثناء يبيح للمسؤول سماع قناة ملكية خاصة بغيره لمجرد أن له حقاً منفصلاً في عرض الطلب إدارياً.

مثال بسيط: زبونان يعرفان رقم ٤٢. بيانات القناة مطابقة في الطلبين لكن الخادم يربط الرقم في كل مرة بهوية مختلفة؛ لذلك يقبل مالك الطلب ويرفض الآخر بلا حاجة لإخفاء الرقم.

  private attach(id: symbol, subscription: Subscription) {
    const session = this.session
    if (!session || !subscription.isAllowed(session)) return
    if (!this.client) {
      try {
        this.client = this.createClient(session)
      } catch {
        console.warn("Real-time connection could not be initialized")
        return
      }
    }
    if (!this.client) return

    let group = this.channels.get(subscription.channel)
    if (!group) {
      const channel = this.client.private(subscription.channel)
      group = {
        channel,
        events: new Map(),
        subscribed: false,
        removeSubscriptionListener: () => {},
      }
      const currentGroup = group
      this.channels.set(subscription.channel, group)
      group.removeSubscriptionListener = channel.onSubscribed(() => {
        if (this.channels.get(subscription.channel) !== currentGroup || this.session !== session) return
        currentGroup.subscribed = true
        for (const event of currentGroup.events.values()) {
          for (const subscriber of event.subscribers.values()) subscriber.onSubscribed()
        }
      })
    }

    let event = group.events.get(subscription.event)
    if (!event) {
      const currentGroup = group
      const subscribers = new Map<symbol, Subscription>()
      const listener: EventListener = (payload) => {
        if (this.channels.get(subscription.channel) !== currentGroup || this.session !== session) return
        for (const subscriber of subscribers.values()) subscriber.onEvent(payload)
      }
      event = { listener, subscribers }
      group.events.set(subscription.event, event)
      group.channel.listen(subscription.event, listener)
    }
    event.subscribers.set(id, subscription)
    if (group.subscribed) subscription.onSubscribed()
  }

ما تقوله

أفترض الآن أن المسؤول خرج وسجّل زبون دخوله من التبويب نفسه. مقارنة الجلسات تشمل المستخدم والدور والرمز. لذلك لا يُعامل اتصال قديم وكأنه صالح للأبد ما دام عنوان الموقع لم يتغير. عند تغير الجلسة يفصل المدير الاتصال السابق، ويسجل الجلسة الجديدة، ثم يحاول ربط الاشتراكات التي ما زالت مطلوبة في الواجهة.

في attach يبدأ كل ربط بفحص وجود الجلسة وملاءمتها للاشتراك. إذا لم تعد إدارية فلن يحاول تركيب مستمع الإدارة لهذا الزبون. وإذا احتاج إلى قناة جديدة فإنه يفتحها كقناة خاصة، فيعود المسار إلى طلب التفويض الذي رأيناه في البداية. كذلك يتأكد المستمع قبل تمرير حدث متأخر أنه ما زال تابعاً لنفس القناة والجلسة. هذه الحراسة تحمي عرض الواجهة أثناء التنقل بين الحسابات، لكن قرار السماح نفسه يبقى عند الخادم. حذف فحص ملكية Laravel لا يعوّضه تنظيف الاتصالات في React.

هذه نقطة مهمة عند تنقل شخصين على نفس الجهاز. حتى الإذن الصادر بصورة صحيحة قبل لحظات يخص جلسة واتصالاً محددين؛ لا نعيد استعماله لمجرد أن المتصفح نفسه ما زال مفتوحاً.

ما تقوله

في الاختبار الأخير أتابع الطلب نفسه مع تغيير الهوية فقط. الزائر المجهول يطلب قناة الإدارة فيُرفض لغياب المصادقة، ثم نرسل كمستخدم معروف بدور زبون فيُرفض لغياب الدور. المسؤول يحصل على رد يتضمن بيانات التفويض؛ لا نكتفي بالقول إن الخادم أعاد صفحة سليمة. بعد ذلك يطلب صاحب الطلب قناته فيُقبل، ويطلب زبون آخر رقم القناة نفسه فيُرفض. حتى المسؤول لا يُقبل في قناة هذا الطلب ما دام ليس مالكه. بهذه المقارنة نعزل شرط الدور عن شرط الملكية.

تُنشئ معظم هذه الحالات مستخدماً مصادقاً عليه مباشرة حتى تختبر قواعد السماح والرفض. لذلك توجد دالة ثانية لها هدف مختلف: تصدر رمزاً فعلياً للمسؤول، تزيل المستخدم المتذكر داخل الحارس، ثم ترسل طلب التفويض بالرمز نفسه. هذا يبيّن ما الذي يفحصه كل اختبار، ولا نخلط نجاح قرار الصلاحية مع تجربة واجهة WebSocket كاملة. أمام الدكتور يكفيني أن أتتبع الطريق بعكسه: قرار القناة جاء من Laravel اعتماداً على هوية الحارس، والمتصفح استعمل الإذن فقط لإكمال اشتراكه المحدد.

لو كان الاختبار يكتفي بقبول المالك لما اكتشف قاعدة تسمح للجميع. ولو اكتفى برفض الغريب لما اكتشف قاعدة ترفض حتى المالك. الجمع بين الحالتين هو ما يجعل المثال مقنعاً.
