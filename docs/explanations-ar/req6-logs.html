١. رحلة معرّف التتبع

يصل طلب الشراء إلى Laravel مع X-Correlation-ID. يحتفظ Laravel بالقيمة مع الطلب، ويمرّرها إلى المستودع في gRPC metadata. عند حفظ الطلب تُخزَّن القيمة في حمولة Outbox. ينشرها العامل الناشر إلى RabbitMQ، ثم يأخذها عامل البريد من الرسالة نفسها. بهذه الطريقة تبقى القيمة موجودة حتى إذا أُوقف عامل البريد وعاد بعد انتهاء طلب HTTP.

الحقلالدور
correlation_idيربط طلب HTTP بما فعله المستودع وما حصل في المعالجة غير المتزامنة.
event_idيربط صف Outbox بالرسالة التي نشرها RabbitMQ وعالجها العامل، ويبقى ثابتاً عند إعادة التسليم.
order_idرقم الطلب الذي يمكن مطابقته مع رد 201 ومع رسالة Mailpit.

افتح backend/my-project/app/Http/Middleware/TraceRequest.php · الأسطر 1–30

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Symfony\Component\HttpFoundation\Response;

class TraceRequest
{
    public function handle(Request $request, Closure $next): Response
    {
        $incoming = $request->header('X-Correlation-ID');
        $id = is_string($incoming) && Str::isUuid($incoming) ? strtolower($incoming) : (string) Str::uuid();
        $request->attributes->set('correlation_id', $id);
        $observed = config('logging.default') === 'observability';
        if ($observed) {
            Log::info('HTTP request started', ['correlation_id' => $id]);
        }
        $response = $next($request);
        $response->headers->set('X-Correlation-ID', $id);
        if ($observed) {
            Log::info('HTTP request finished', ['correlation_id' => $id, 'status' => $response->getStatusCode()]);
        }

        return $response;
    }
}

هذا وسيط جديد. header يقرأ القيمة الواردة. Str::isUuid يتحقق من الصيغة وstrtolower يوحّدها. إذا كانت القيمة مفقودة أو غير صالحة، يختار Str::uuid() قيمة جديدة. تخزّن attributes القيمة داخل كائن الطلب حتى يستطيع أي كود لهذا الطلب قراءتها. يكتب الشرط $observed حدثين عند استخدام قناة المراقبة. السطر $next($request) يشغّل بقية Laravel. عقب رجوع الرد يضيف الوسيط الترويسة ويكتب حالة HTTP مثل 201.

    public function formatBatch(array $records): string

    public function format(LogRecord $record): string
    {
        $row = ['time' => $record->datetime->format('Y-m-d\TH:i:s.uP'), 'level' => $record->level->getName(),
            'message' => in_array($record->message, self::EVENTS, true) ? $record->message : 'Application diagnostic'];
        foreach (['event_id', 'correlation_id'] as $key) {
            $value = $record->context[$key] ?? null;
            if (is_string($value) && Str::isUuid($value)) {
                $row[$key] = $value;
            }
        }
        foreach (['order_id', 'attempt', 'status', 'operation_id', 'compensation_id'] as $key) {
            if (isset($record->context[$key]) && is_int($record->context[$key])) {
                $row[$key] = $record->context[$key];
            }
        }
        foreach (['delivery_mode' => ['external', 'mock'], 'destination' => ['retry', 'dead_letter']] as $key => $values) {
            if (in_array($record->context[$key] ?? null, $values, true)) {
                $row[$key] = $record->context[$key];
            }
        }
        if (($record->context['exception'] ?? null) instanceof Throwable) {
            $row['exception_type'] = $record->context['exception']::class;
        }

        return json_encode($row, JSON_THROW_ON_ERROR)."\n";
    }
}

formatBatch يمر على السجلات بالدالة نفسها. يبدأ format بوقت ISO مع أجزاء الثانية ومستوى السجل واسم الحدث، ثم يضيف الحقول المسموحة. يفحص UUID للمعرّفات، والنوع الصحيح للأرقام، والقيم المعروفة للحقول النصية المحددة. ينهي json_encode كل حدث بسطر جديد حتى يستطيع برنامج القراءة تحليل كل سطر مستقلاً.

٣. المستودع · المعرّف يعبر بصيغة gRPC metadata

افتح backend/my-project/app/Services/InventoryGrpcClient.php · الأسطر 80–109

    private function call(string $method, array $request): array
    {
        $client = $this->client();
        $trace = request()->attributes->get('correlation_id');
        $metadata = $trace ? ['x-correlation-id' => [$trace]] : [];
        $call = $client->{$method}($request, $metadata, ['timeout' => $this->timeoutMicros]);
        [$response, $status] = $call->wait();

        if ($status->code !== \Grpc\STATUS_OK) {
            if ($status->code === \Grpc\STATUS_DEADLINE_EXCEEDED && $method === 'ReserveStock') {
                throw new InventoryOutcomeUnknownException('Inventory request deadline exceeded.');
            }

            if ($status->code === \Grpc\STATUS_UNAVAILABLE || $status->code === \Grpc\STATUS_DEADLINE_EXCEEDED) {
                throw new InventoryUnavailableException('Inventory service is unavailable.');
            }

            throw new InventoryUnavailableException('Inventory request failed.');
        }

        if ($response instanceof InventoryProtoResponse) {
            $response = $response->toArray();
        }

        if (! is_array($response)) {
            throw new InventoryUnavailableException('Inventory service returned an invalid response.');
        }

        return $response;
    }

عدّلنا الدالة المشتركة call() لأنها النقطة التي تمر منها عمليات GetStock وReserveStock وReleaseStock. تقرأ القيمة من الطلب الحالي وتضعها باسم x-correlation-id كمصفوفة قيم في الوسيط الثاني لاستدعاء stub. بهذا يبقى تعريف رسائل protobuf كما هو، ويصل المعرّف إلى الخدمة خارج جسم طلب المخزون. يظل تحويل أخطاء gRPC والتحقق من نتيجة الرد في نفس الدالة.

افتح backend/my-project/app/Services/Messaging/OutboxPublisher.php · الأسطر 14–52

    public function publishNext(): bool
    {
        return DB::transaction(function (): bool {
            $event = OutboxEvent::query()->useWritePdo()
                ->whereNull('published_at')->where('available_at', '<=', now())
                ->orderBy('available_at')->orderBy('event_id')->lockForUpdate()->first();
            if ($event === null) {
                return false;
            }

            $context = ['event_id' => $event->event_id, 'order_id' => $event->order_id,
                'correlation_id' => $event->payload['correlation_id'] ?? $event->event_id];
            try {
                $this->broker->publish(
                    json_encode($event->payload, JSON_THROW_ON_ERROR),
                    $event->event_id,
                    config('rabbitmq.exchange'),
                    config('rabbitmq.routing_key'),
                );
            } catch (\Throwable $exception) {
                $attempts = $event->attempts + 1;
                $event->update([
                    'attempts' => $attempts,
                    'available_at' => now()->addSeconds(min(config('rabbitmq.publisher_max_backoff'), 2 ** min($attempts, 12))),
                    'last_error' => $exception::class,
                ]);
                $this->broker->close();
                Log::warning('Outbox publication deferred', $context + ['attempt' => $attempts, 'error_type' => $exception::class]);

                return true;
            }

            // A crash after the confirm but before this commit republishes the same event ID.
            $event->update(['published_at' => now(), 'attempts' => $event->attempts + 1, 'last_error' => null]);
            Log::info('Outbox publication confirmed', $context);

            return true;
        }, 1);
    }

أضفنا بيانات السياق بعد قراءة الصف المقفل. يستخدم سجل التأجيل والسجل الذي يؤكد النشر السياق نفسه. يُنشر JSON الحمولة ومعرّف الحدث، وبعد قبول الوسيط يحدّث الناشر published_at ويسجل Outbox publication confirmed. الأحداث الأقدم التي لا تحمل معرّف طلب HTTP يمكن ربط ناشرها وعاملها بواسطة UUID الحدث.

افتح backend/my-project/app/Services/Messaging/EmailMessageProcessor.php · الأسطر 13–58

    public function handle(IncomingMessage $message): void
    {
        $event = null;
        $trace = [];
        try {
            $event = $this->decode($message->body);
            $trace = ['correlation_id' => $event['correlation_id'] ?? $event['event_id'], 'order_id' => $event['order_id']];
            Log::info('Order email processing', ['event_id' => $event['event_id'], 'attempt' => $message->attempt] + $trace);
            $this->sender->send($event);
        } catch (MailDeliveryConfigurationException $exception) {
            // Misconfiguration is operational, not a failed delivery attempt. Preserve the original message.
            throw $exception;
        } catch (\Throwable $exception) {
            $permanent = $exception instanceof PermanentMessageException || $exception instanceof JsonException;
            $exhausted = $message->attempt >= config('rabbitmq.max_attempts');
            $eventId = $event['event_id'] ?? (string) Str::uuid();
            if ($permanent || $exhausted) {
                // Confirm the DLQ publication before acknowledging the original delivery.
                $this->broker->publish($message->body, $eventId, config('rabbitmq.dead_letter_exchange'), config('rabbitmq.dead_letter_routing_key'), $message->attempt);
            } else {
                $retryIndex = min($message->attempt, count(config('rabbitmq.retry_delays')));
                $this->broker->publish($message->body, $eventId, config('rabbitmq.retry_exchange'), 'retry.'.$retryIndex, $message->attempt + 1);
            }
            Log::warning('Order email deferred', [
                'event_id' => $eventId, 'attempt' => $message->attempt,
                'destination' => $permanent || $exhausted ? 'dead_letter' : 'retry', 'error_type' => $exception::class,
            ] + $trace);
            $message->acknowledge();

            return;
        }

        Log::info('Order email processed', ['event_id' => $event['event_id'], 'attempt' => $message->attempt] + $trace);
        $message->acknowledge();
    }

    private function decode(string $body): array
    {
        $event = json_decode($body, true, 16, JSON_THROW_ON_ERROR);
        if (! is_array($event) || ($event['type'] ?? null) !== 'OrderCreated'
            || ! is_string($event['event_id'] ?? null) || ! Str::isUuid($event['event_id'])
            || ! is_int($event['order_id'] ?? null) || $event['order_id'] < 1) {
            throw new PermanentMessageException('Unsupported order event envelope.');
        }

        return $event;

بعد قراءة الرسالة، يتحقق decode من الظرف ويستخرج العامل الحقول. يبني $trace داخل handle لكل رسالة. تُضمّ الحقول إلى سجلات البدء والتأجيل والإنهاء. عند إعادة المحاولة تبقى حمولة الرسالة الأصلية، ويزيد رقم المحاولة. في المسار الناجح يسجل العامل الإنجاز ثم يؤكّد الرسالة.

افتح backend/my-project/app/Services/Messaging/OrderConfirmationSender.php · الأسطر 18–55

    public function send(array $event): void
    {
        $mode = $this->mailConfiguration->assertDeliverable();

        $outbox = OutboxEvent::query()->useWritePdo()->find($event['event_id']);
        if ($outbox === null || $outbox->event_type !== 'OrderCreated' || $outbox->order_id !== $event['order_id']) {
            throw new PermanentMessageException('Message does not identify a recorded order event.');
        }

        // Unique row + row lock serializes duplicate deliveries, even across worker processes.
        DB::table('notification_deliveries')->insertOrIgnore([
            'event_id' => $event['event_id'], 'order_id' => $event['order_id'],
            'created_at' => now(), 'updated_at' => now(),
        ]);
        DB::transaction(function () use ($event, $mode): void {
            $delivery = NotificationDelivery::query()->whereKey($event['event_id'])->lockForUpdate()->firstOrFail();
            if ($delivery->delivered_at !== null) {
                // A provider change does not automatically replay earlier mock or unclassified deliveries.
                return;
            }

            $order = Order::query()->useWritePdo()->with(['user', 'items.product'])->find($event['order_id']);
            if ($order === null || $order->user === null || ! filter_var($order->user->email, FILTER_VALIDATE_EMAIL)) {
                throw new PermanentMessageException('Order recipient is unavailable.');
            }

            $sent = Mail::to($order->user->email)->send(new OrderConfirmation($order, $event['event_id']));
            if ($sent === null) {
                throw new RuntimeException('Mail delivery was cancelled before sending.');
            }
            // SMTP cannot participate in this transaction. A crash here can resend; stable Message-ID helps providers deduplicate.
            $delivery->update(['delivered_at' => now(), 'delivery_mode' => $mode]);
            Log::info($mode === 'mock' ? 'MOCK DELIVERY: order confirmation processed without an external email' : 'Order confirmation accepted by delivery transport', [
                'event_id' => $event['event_id'], 'delivery_mode' => $mode,
                'transport_message_id' => substr(preg_replace('/[^A-Za-z0-9@._:+\/-]/', '', $sent->getMessageId()), 0, 255),
            ] + array_intersect_key($event, ['correlation_id' => true]));
        }, 1);
    }

عدّلنا سجل قبول ناقل البريد ليحمل مفتاح التتبع الموجود في الحدث، باستعمال array_intersect_key لاختيار ذلك المفتاح. يرى الأستاذ ترتيب processing ثم قبول النقل ثم processed. يمكنه مطابقة رقم الطلب في Mailpit مع رقم السجل. شرط الصف المقفل يتخطى إرسال حدث سبق تسليمه.

٥. تجربة أمام الأستاذ · أظهر السجل الخام والتطابق

  2. يجمع compose.sh ملفات البنية وإعداد المقابلة والمراقبة ضمن مشروع ecommerce-req6. يفعّل --local-sources ربط الكود الحالي على صور العرض الجاهزة. يشغّل up -d الخدمات في الخلفية، ويطلب --no-build --pull never استخدام الصور المحلية، وينتظر --wait جاهزية الحاويات.
