القيم الافتراضية تحسب أول محاولة ضمن الخمس. إذا فشل الإرسال مؤقتاً في المرة الأولى نؤجله خمس ثوانٍ، ثم ثلاثين، ثم مئة وعشرين، ثم ثلاثمئة. إذا فشلت المحاولة الخامسة فلا توجد سادسة. هذه مواعيد لإعادة تجربة خطأ قابل للتجاوز، وليست مدة انتظار يفرضها العامل على الزبون. الرسالة التالفة، بالمقابل، لن تصبح صالحة بعد خمس ثوانٍ، ولهذا لها مسار آخر.

لتثبيت الفكرة أتتبع طلباً واحداً فقط: رقم الطلب ثابت، والتأخير ورقم المحاولة يخصان العمل على تأكيده. لا ننشئ طلب شراء جديداً عند الانتقال بين هذه الطوابير.

    private function declareTopology(AMQPChannel $channel): void
    {
        foreach (['exchange', 'retry_exchange', 'dead_letter_exchange'] as $key) {
            $channel->exchange_declare(config('rabbitmq.'.$key), 'direct', false, true, false);
        }

        $quorum = ['x-queue-type' => 'quorum'];
        $channel->queue_declare(config('rabbitmq.dead_letter_queue'), false, true, false, false, false, new AMQPTable($quorum));
        $channel->queue_bind(config('rabbitmq.dead_letter_queue'), config('rabbitmq.dead_letter_exchange'), config('rabbitmq.dead_letter_routing_key'));

        $channel->queue_declare(config('rabbitmq.queue'), false, true, false, false, false, new AMQPTable($quorum + [
            'x-dead-letter-exchange' => config('rabbitmq.dead_letter_exchange'),
            'x-dead-letter-routing-key' => config('rabbitmq.dead_letter_routing_key'),
            'x-dead-letter-strategy' => 'at-least-once',
            'x-overflow' => 'reject-publish',
        ]));
        $channel->queue_bind(config('rabbitmq.queue'), config('rabbitmq.exchange'), config('rabbitmq.routing_key'));

        foreach (config('rabbitmq.retry_delays') as $index => $seconds) {
            $queue = config('rabbitmq.queue').'.retry.'.($index + 1);
            $channel->queue_declare($queue, false, true, false, false, false, new AMQPTable($quorum + [
                'x-message-ttl' => $seconds * 1000,
                'x-dead-letter-exchange' => config('rabbitmq.exchange'),
                'x-dead-letter-routing-key' => config('rabbitmq.routing_key'),
                'x-dead-letter-strategy' => 'at-least-once',
                'x-overflow' => 'reject-publish',
            ]));
            $channel->queue_bind($queue, config('rabbitmq.retry_exchange'), 'retry.'.($index + 1));
        }
    }

ما تقوله

في declareTopology أقرأ أسماء الخيارات لأنها تشرح الآلية. الـexchange موزّع يختار وجهة باستخدام routing key، أما الـqueue فهي التي تحتفظ بالرسالة. لهذا نصرّح بطابور الفشل ثم نربطه بموزّعه ومفتاحه؛ إنشاء طابور بلا هذا الربط لا يجعل رسائل الفشل تصل إليه تلقائياً. الخيار durable في تعريف الطابور يطلب بقاء تعريفه بعد إعادة تشغيل الوسيط مع بقاء بياناته. ونستخدم نوع quorum؛ هذا وحده لا يعني أن عرضنا المحلي مشغّل على عدة وسطاء.

انظر الآن إلى طابور الانتظار الأول. قيمة x-message-ttl هي عمر الرسالة بالميلي ثانية، ولذلك يتحول تأخير خمس ثوانٍ إلى خمسة آلاف. وعند انقضاء الوقت يحدد x-dead-letter-exchange ورفيقه الخاص بمفتاح التوجيه رحلة العودة إلى المسار العادي. كلمة dead-letter هنا لا تعني أن كل رسالة منتهية ذاهبة إلى طابور الفشل النهائي؛ في طابور الانتظار ضبطنا الوجهة لتكون موزّع العمل العادي. لذلك يستطيع العامل معالجة رسائل أخرى أثناء الانتظار.

بهذا التوجيه تعود رسالة مهلة الخمس ثواني إلى نفس مسار بقية رسائل البريد. لا يلزم عامل متخصص بكل مدة، ولا نحجز عاملاً ليظل نائماً فوق رسالة واحدة.

    public function receive(): ?IncomingMessage
    {
        // Polling keeps heartbeat/socket handling bounded, including on an idle queue.
        $message = $this->channel()->basic_get(config('rabbitmq.queue'), false);
        if ($message === null) {
            return null;
        }
        $headers = $message->has('application_headers') ? $message->get('application_headers')->getNativeData() : [];
        $attempt = filter_var($headers['attempt'] ?? 1, FILTER_VALIDATE_INT);

        return new IncomingMessage(
            $message->getBody(),
            $attempt === false ? 1 : max(1, $attempt),
            fn () => $message->ack(),
        );
    }

    public function handle(MessageBroker $broker, EmailMessageProcessor $processor, MailDeliveryConfiguration $mailConfiguration): int
    {
        try {
            // Check before connecting to the database or consuming any message.
            $mode = $mailConfiguration->assertDeliverable();
        } catch (MailDeliveryConfigurationException $exception) {
            $this->error($exception->getMessage());

            return self::FAILURE;
        }
        if ($mode === 'mock') {
            $this->warn('MOCK DELIVERY enabled: order confirmations will not send external email.');
            Log::warning('MOCK DELIVERY enabled for order email worker; no external email will be sent');
        }
        DB::connection()->useWriteConnectionWhenReading();
        do {
            try {
                $message = $broker->receive();
                if ($message !== null) {
                    $processor->handle($message);
                } elseif (! $this->option('once')) {
                    sleep(1);
                }
            } catch (MailDeliveryConfigurationException $exception) {
                $broker->close();
                $this->error($exception->getMessage());

                return self::FAILURE;
            } catch (\Throwable $exception) {
                // Closing leaves the delivery unacknowledged; RabbitMQ redelivers on recovery.
                $broker->close();
                Log::error('Order email worker recovering', ['error_type' => $exception::class]);
                if ($this->option('once')) {
                    return self::FAILURE;
                }
                sleep(config('rabbitmq.reconnect_delay'));
            }
        } while (! $this->option('once'));
        $broker->close();

        return self::SUCCESS;
    }

ما تقوله

في receive الفكرة ليست السحب فقط، بل توقيت حذف مسؤولية الوسيط. المعامل الثاني لـbasic_get هو no_ack، ووضع false يعني أن الإقرار يدوي. يحصل العامل على جسم الرسالة ومعه وسيلة لتنفيذ الإقرار لاحقاً. لا تُستدعى تلك الوسيلة بمجرد تكوين النتيجة. إذا تعطّل العامل الآن، بعد الاستلام وقبل الإقرار، يستطيع الوسيط إعادة تسليم الأصل بعد انقطاع الاتصال.

في حلقة العامل أفرّق بين ثلاث لحظات: لو كان متوقفاً قبل السحب تبقى الرسالة جاهزة؛ ولو كان يحملها ولم يؤكدها تكون غير مُقرّ بها؛ ولو أنهى العمل وأكدها لا نطلب إعادتها بسبب إعادة تشغيله. إغلاق الاتصال عند خطأ غير متوقع يترك ما لم يُقَرّ به قابلاً للتسليم مجدداً. هذا ليس نفس مسار إعادة المحاولة المؤجلة: إعادة التسليم بسبب انقطاع العامل وحدها لا تعني أننا زدنا ترويسة رقم المحاولة.

أفضل سؤال عند توقف الخدمة ليس «هل كانت الحاوية شغالة؟»، بل «هل تأكد الوسيط أن هذه الرسالة انتهت؟». مكان التوقف بالنسبة للإقرار هو الذي يغير الجواب.

    public function handle(IncomingMessage $message): void
    {
        $event = null;
        try {
            $event = $this->decode($message->body);
            Log::info('Order email processing', ['event_id' => $event['event_id'], 'attempt' => $message->attempt]);
            $this->sender->send($event);
        } catch (MailDeliveryConfigurationException $exception) {
            // Misconfiguration is operational, not a failed delivery attempt. Preserve the original message.
            throw $exception;
        } catch (\Throwable $exception) {
            $permanent = $exception instanceof PermanentMessageException || $exception instanceof JsonException;
            $exhausted = $message->attempt >= config('rabbitmq.max_attempts');
            $eventId = $event['event_id'] ?? (string) Str::uuid();
            if ($permanent || $exhausted) {
                // Confirm the DLQ publication before acknowledging the original delivery.
                $this->broker->publish($message->body, $eventId, config('rabbitmq.dead_letter_exchange'), config('rabbitmq.dead_letter_routing_key'), $message->attempt);
            } else {
                $retryIndex = min($message->attempt, count(config('rabbitmq.retry_delays')));
                $this->broker->publish($message->body, $eventId, config('rabbitmq.retry_exchange'), 'retry.'.$retryIndex, $message->attempt + 1);
            }
            Log::warning('Order email deferred', [
                'event_id' => $eventId, 'attempt' => $message->attempt,
                'destination' => $permanent || $exhausted ? 'dead_letter' : 'retry', 'error_type' => $exception::class,
            ]);
            $message->acknowledge();

            return;
        }

        Log::info('Order email processed', ['event_id' => $event['event_id'], 'attempt' => $message->attempt]);
        $message->acknowledge();
    }

ما تقوله

في المعالج أشرح القرار كوحدة واحدة. النجاح واضح: نفهم الحدث، نرسل البريد، ثم نؤكد الأصل. خطأ إعداد البريد حالة مختلفة؛ إذا كانت المشكلة في إعداد الخدمة كله، لا يصح أن نحرق رصيد محاولات كل زبون. لذلك يُرفع الاستثناء للعامل من دون إقرار الأصل. أما بقية الأخطاء فيسأل الكود: هل الحدث تالف أو غير صالح، أم أننا استنفدنا الحد المسموح؟

إذا كان الجواب نعم، ننشر الجسم إلى طابور الفشل بمفتاحه ونحتفظ برقم المحاولة. إذا كان الخطأ مؤقتاً وما زالت هناك فرصة، نختار مرحلة التأخير وننشر نسخة برقم محاولة أعلى. في الفرعين تأتي عملية إقرار الأصل بعد نشر البديل، لا قبله. مثلاً إذا تعطّل الوسيط ونحن نحاول نقل رسالة إلى الانتظار، لا نريد أن نكون قد حذفنا الأصل اعتماداً على نسخة لا نعرف إن كانت وصلت. معنى «لا تضيع» يتوقف على هذا الترتيب.

هذا أيضاً يفسر لماذا نحتاج طابوراً للفشل النهائي: نحتفظ بالحمولة الأصلية لفهم السبب، ولا نستهلك وقت التشغيل بإعادة حدث نعرف أن بياناته الأساسية غير صالحة.

    private function channel(): AMQPChannel
    {
        if ($this->channel !== null && $this->channel->is_open()) {
            return $this->channel;
        }

        $this->connection = new AMQPStreamConnection(
            host: config('rabbitmq.host'),
            port: config('rabbitmq.port'),
            user: config('rabbitmq.user'),
            password: config('rabbitmq.password'),
            vhost: config('rabbitmq.vhost'),
            connection_timeout: config('rabbitmq.connection_timeout'),
            read_write_timeout: config('rabbitmq.read_write_timeout'),
            keepalive: function_exists('socket_import_stream'),
            heartbeat: config('rabbitmq.heartbeat'),
            channel_rpc_timeout: config('rabbitmq.confirm_timeout'),
        );
        $this->channel = $this->connection->channel();
        $this->declareTopology($this->channel);
        $this->channel->confirm_select();
        $this->channel->basic_qos(0, 1, false);

        return $this->channel;
    }

    public function publish(string $body, string $eventId, string $exchange, string $routingKey, int $attempt = 1): void
    {
        $channel = $this->channel();
        $confirmed = false;
        $failed = false;
        $channel->set_ack_handler(function () use (&$confirmed): void {
            $confirmed = true;
        });
        $channel->set_nack_handler(function () use (&$failed): void {
            $failed = true;
        });
        $channel->set_return_listener(function () use (&$failed): void {
            $failed = true;
        });
        $message = new AMQPMessage($body, [
            'content_type' => 'application/json',
            'delivery_mode' => AMQPMessage::DELIVERY_MODE_PERSISTENT,
            'message_id' => $eventId,
            'type' => 'OrderCreated',
            'timestamp' => time(),
            'application_headers' => new AMQPTable(['attempt' => $attempt]),
        ]);
        $channel->basic_publish($message, $exchange, $routingKey, true);
        $channel->wait_for_pending_acks_returns(config('rabbitmq.confirm_timeout'));
        if (! $confirmed || $failed) {
            throw new RuntimeException('RabbitMQ did not confirm a routed message.');
        }
    }

ما تقوله

هنا يوجد تأكيدان متعاكسان، وأشير إلى كل منهما باسمه. تفعيل confirm_select يجعل RabbitMQ يخبر الناشر بنتيجة نشر البديل. أما ACK الذي شرحناه سابقاً فيرسله المستهلك للوسيط ليقول إنه انتهى من الأصل. الأول يشهد على النشر، والثاني يسمح بالتخلي عن الرسالة المستلمة؛ لا يجوز أن يحل أحدهما محل الآخر.

في publish نطلب حفظ الرسالة بشكل دائم ونرسلها مع mandatory مفعّلاً لكي لا نعتبر رسالة غير قابلة للتوجيه نجاحاً. ننتظر ردود التأكيد والإرجاع ضمن مهلة، ونرفع استثناءً إذا غاب التأكيد أو ظهر رفض أو إرجاع. لذلك لا يصل المعالج إلى إقرار الأصل حين لا يثبت نجاح النقل. والعكس ممكن أيضاً: قد يصل البديل ثم يتوقف العامل قبل إقرار الأصل، فتظهر نسخة مكررة. نقبل احتمال التكرار لأنه أفضل من ضياع تأكيد الشراء.

لاحظ أن معرّف الحدث يبقى كما هو في النقل، بينما ترويسة المحاولة وحدها تتقدم عند إعادة المحاولة المبرمجة. نستطيع بذلك الربط بين المراحل المختلفة لنفس التأكيد.

    public function send(array $event): void
    {
        $mode = $this->mailConfiguration->assertDeliverable();

        $outbox = OutboxEvent::query()->useWritePdo()->find($event['event_id']);
        if ($outbox === null || $outbox->event_type !== 'OrderCreated' || $outbox->order_id !== $event['order_id']) {
            throw new PermanentMessageException('Message does not identify a recorded order event.');
        }

        // Unique row + row lock serializes duplicate deliveries, even across worker processes.
        DB::table('notification_deliveries')->insertOrIgnore([
            'event_id' => $event['event_id'], 'order_id' => $event['order_id'],
            'created_at' => now(), 'updated_at' => now(),
        ]);
        DB::transaction(function () use ($event, $mode): void {
            $delivery = NotificationDelivery::query()->whereKey($event['event_id'])->lockForUpdate()->firstOrFail();
            if ($delivery->delivered_at !== null) {
                // A provider change does not automatically replay earlier mock or unclassified deliveries.
                return;
            }

            $order = Order::query()->useWritePdo()->with(['user', 'items.product'])->find($event['order_id']);
            if ($order === null || $order->user === null || ! filter_var($order->user->email, FILTER_VALIDATE_EMAIL)) {
                throw new PermanentMessageException('Order recipient is unavailable.');
            }

            $sent = Mail::to($order->user->email)->send(new OrderConfirmation($order, $event['event_id']));
            if ($sent === null) {
                throw new RuntimeException('Mail delivery was cancelled before sending.');
            }
            // SMTP cannot participate in this transaction. A crash here can resend; stable Message-ID helps providers deduplicate.
            $delivery->update(['delivered_at' => now(), 'delivery_mode' => $mode]);
            Log::info($mode === 'mock' ? 'MOCK DELIVERY: order confirmation processed without an external email' : 'Order confirmation accepted by delivery transport', [
                'event_id' => $event['event_id'], 'delivery_mode' => $mode,
                'transport_message_id' => substr(preg_replace('/[^A-Za-z0-9@._:+\/-]/', '', $sent->getMessageId()), 0, 255),
            ]);
        }, 1);
    }

آخر جزء يفسر ما يفعله النظام عندما تصل الواقعة نفسها مرتين. معرّف الحدث ثابت، ولدينا سجل تسليم مرتبط به في قاعدة البيانات. يأخذ العامل قفلاً على ذلك السجل، ثم ينظر إلى وقت التسليم. إذا كان مسجلاً من معالجة سابقة يرجع بلا إرسال جديد؛ فلا يتحول كل تكرار على الطابور تلقائياً إلى رسالة ثانية في بريد الزبون.

الحد المهم أمام الدكتور هو المسافة بين استدعاء مزوّد البريد وتثبيت نتيجة الإرسال محلياً. لو قبل المزوّد الرسالة ثم تعطلت العملية قبل حفظ الوقت، لا نملك معاملة واحدة تشمل المزوّد وMySQL. لذلك الوصف الدقيق هو أن توقف العامل قبل الإقرار لا يُسقط الرسالة، وأن عندنا حماية من الإرسال المكرر في المعالجات المسجلة. لا نستنتج أن البريد الخارجي سيخرج مرة واحدة بالضبط في كل توقيت ممكن للانهيار.

فإذا رجع العامل بعد دقيقة مثلاً، لا نقول له أرسل كل طلبات آخر دقيقة يدوياً؛ يكمل ما يسلّمه الوسيط ويستعين بسجل التسليم لمعرفة ما أتمّه سابقاً.
